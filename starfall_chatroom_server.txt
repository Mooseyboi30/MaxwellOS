local function isWordByte(byte)
    return byte ~= nil and ((byte >= 97 and byte <= 122) or (byte >= 48 and byte <= 57))
end

local function cleanChatText(value)
    local chars = {}
    local first, last
    for i = 1, #value do
        local byte = value:byte(i)
        if byte <= 32 or byte == 127 then
            chars[i] = " "
        else
            chars[i] = value:sub(i, i)
            first, last = first or i, i
        end
    end
    return first and table.concat(chars, "", first, last) or ""
end

local function containsSlur(message)
    local lower = message:lower()
    for start = 1, #lower do
        local words = blockedByInitial[lower:byte(start)]
        if words and (start == 1 or not isWordByte(lower:byte(start - 1))) then
            for _, word in ipairs(words) do
                local pos, letter = start + 1, 2
                while letter <= #word do
                    while pos <= #lower and not isWordByte(lower:byte(pos)) do pos = pos + 1 end
                    if lower:byte(pos) ~= word:byte(letter) then break end
                    pos, letter = pos + 1, letter + 1
                end
                if letter > #word then
                    if lower:byte(pos) == 115 then pos = pos + 1 end
                    if not isWordByte(lower:byte(pos)) then return true end
                end
            end
        end
    end
    return false
end

local function containsSlurInName(name)
    local compact = name:lower():gsub("%W", "")
    for _, word in ipairs(blockedWords) do
        if compact:find(word, 1, true) then return true end
    end
    return false
end

local CHESS_START = "rnbqkbnrpppppppp................................PPPPPPPPRNBQKBNR"
local PROMOTIONS = {"q", "r", "b", "n"}

local function pieceAt(board, square)
    return board:sub(square, square)
end

local function setPiece(board, square, piece)
    return board:sub(1, square - 1) .. piece .. board:sub(square + 1)
end

local function isWhite(piece)
    return piece ~= "." and piece == piece:upper()
end

local function geometry(board, from, to, piece, attacksOnly)
    local dx = (to - 1) % 8 - (from - 1) % 8
    local dy = math.floor((to - 1) / 8) - math.floor((from - 1) / 8)
    local ax, ay = math.abs(dx), math.abs(dy)
    local kind = piece:lower()
    if from == to then return false end
    if kind == "p" then
        local direction = isWhite(piece) and -1 or 1
        if attacksOnly then return ax == 1 and dy == direction end
        if dx ~= 0 or pieceAt(board, to) ~= "." then return false end
        if dy == direction then return true end
        local start = isWhite(piece) and 6 or 1
        return dy == direction * 2 and math.floor((from - 1) / 8) == start
            and pieceAt(board, from + direction * 8) == "."
    end
    if kind == "n" then return ax * ay == 2 end
    if kind == "k" then return ax <= 1 and ay <= 1 end
    local straight, diagonal = dx == 0 or dy == 0, ax == ay
    if not ((kind == "r" and straight) or (kind == "b" and diagonal)
        or (kind == "q" and (straight or diagonal))) then return false end
    local step = (dx == 0 and 0 or (dx > 0 and 1 or -1)) + (dy == 0 and 0 or (dy > 0 and 8 or -8))
    for square = from + step, to - step, step do
        if pieceAt(board, square) ~= "." then return false end
    end
    return true
end

local function attacked(board, square, byWhite)
    for from = 1, 64 do
        local piece = pieceAt(board, from)
        if piece ~= "." and isWhite(piece) == byWhite and geometry(board, from, square, piece, true) then return true end
    end
    return false
end

local function inCheck(board, white)
    return attacked(board, board:find(white and "K" or "k", 1, true), not white)
end

local function chessMove(game, from, to, white, promotion)
    local board = game.board
    local piece, target = pieceAt(board, from), pieceAt(board, to)
    if piece == "." or isWhite(piece) ~= white or (target ~= "." and isWhite(target) == white)
        or target:lower() == "k" then return end
    local kind = piece:lower()
    local dx = (to - 1) % 8 - (from - 1) % 8
    local rookFrom, rookTo, capture
    local home = white and 61 or 5
    if kind == "k" and from == home and (to == home + 2 or to == home - 2) then
        local step = to > from and 1 or -1
        local right = step == 1 and "k" or "q"
        if white then right = right:upper() end
        if not game.rights:find(right, 1, true) then return end
        rookFrom, rookTo = home + (step == 1 and 3 or -4), home + step
        if pieceAt(board, rookFrom) ~= (white and "R" or "r") then return end
        for square = from + step, rookFrom - step, step do
            if pieceAt(board, square) ~= "." then return end
        end
        local crossing = setPiece(setPiece(board, from, "."), rookTo, piece)
        if inCheck(board, white) or attacked(crossing, rookTo, not white) then return end
    elseif kind == "p" and dx ~= 0 then
        if not geometry(board, from, to, piece, true) then return end
        if target == "." then
            if to ~= game.enPassant then return end
            capture = to + (white and 8 or -8)
            if pieceAt(board, capture) ~= (white and "p" or "P") then return end
        end
    elseif not geometry(board, from, to, piece, false) then
        return
    end

    local placed = piece
    if kind == "p" and (to <= 8 or to > 56) then
        placed = white and promotion:upper() or promotion
    end
    board = setPiece(setPiece(board, from, "."), to, placed)
    if capture then board = setPiece(board, capture, ".") end
    if rookFrom then board = setPiece(setPiece(board, rookFrom, "."), rookTo, white and "R" or "r") end
    if inCheck(board, white) then return end
    return board
end

local FLEET_LENGTHS = {5, 4, 3, 3, 2}
local SHIP_NAMES = {"Carrier", "Battleship", "Cruiser", "Submarine", "Destroyer"}
local FLEET_CELLS = 17

local function fleetsReady(game)
    return game.guest ~= nil and #game.placements.X == #FLEET_LENGTHS and #game.placements.O == #FLEET_LENGTHS
end

local function placeShip(game, mark, value)
    if fleetsReady(game) then return "Both fleets are ready. Placement is locked." end
    local ships, fleet = game.placements[mark], game.fleets[mark]
    if value == 301 then
        local last = table.remove(ships)
        if not last then return "No ships to undo." end
        for i = 0, last.length - 1 do fleet[last.square + i * last.step] = nil end
        return "ok"
    end
    local length = FLEET_LENGTHS[#ships + 1]
    if not length then return "Your fleet is already placed." end
    local step = value > 200 and 10 or 1
    local square = value - (step == 10 and 200 or 100)
    local edge = step == 1 and (square - 1) % 10 or math.floor((square - 1) / 10)
    if edge + length > 10 then return "That ship goes off the grid." end
    for i = 0, length - 1 do
        if fleet[square + i * step] then return "Ships cannot overlap." end
    end
    local ship = {square = square, step = step, length = length, name = SHIP_NAMES[#ships + 1], hits = 0}
    for i = 0, length - 1 do fleet[square + i * step] = ship end
    ships[#ships + 1] = ship
    return "ok"
end

local function battleshipBoards(game, mark)
    local opponent = mark == "X" and "O" or "X"
    local target, own = {}, {}
    for square = 1, 100 do
        target[square] = game.shots[mark][square] or "."
        own[square] = game.shots[opponent][square] or (game.fleets[mark][square] and "S" or ".")
    end
    return table.concat(target), table.concat(own)
end

local function fileName(square)
    return string.char(97 + (square - 1) % 8)
end

local function rankName(square)
    return tostring(8 - math.floor((square - 1) / 8))
end

local function chessNotation(game, from, to, white, promotion)
    local board = game.board
    local piece = pieceAt(board, from)
    local kind = piece:upper()
    if kind == "K" and math.abs(to - from) == 2 then return to > from and "O-O" or "O-O-O" end
    local capture = pieceAt(board, to) ~= "." or (kind == "P" and fileName(from) ~= fileName(to))
    local prefix = kind == "P" and (capture and fileName(from) or "") or kind
    if kind ~= "P" and kind ~= "K" then
        local ambiguous, sameFile, sameRank = false, false, false
        for candidate = 1, 64 do
            if candidate ~= from and pieceAt(board, candidate) == piece
                and chessMove(game, candidate, to, white, "q") then
                ambiguous = true
                sameFile = sameFile or fileName(candidate) == fileName(from)
                sameRank = sameRank or rankName(candidate) == rankName(from)
            end
        end
        if ambiguous then
            if not sameFile then prefix = prefix .. fileName(from)
            elseif not sameRank then prefix = prefix .. rankName(from)
            else prefix = prefix .. fileName(from) .. rankName(from) end
        end
    end
    local suffix = kind == "P" and (to <= 8 or to > 56) and ("=" .. promotion:upper()) or ""
    return prefix .. (capture and "x" or "") .. fileName(to) .. rankName(to) .. suffix
end

local function hasChessMove(game, white)
    for from = 1, 64 do
        local piece = pieceAt(game.board, from)
        if piece ~= "." and isWhite(piece) == white then
            for to = 1, 64 do
                if chessMove(game, from, to, white, "q") then return true end
            end
        end
    end
    return false
end

local function playChess(game, value, mark)
    local from = math.floor(value / 256)
    local to = math.floor(value / 4) % 64 + 1
    if from < 1 or from > 64 then return "Choose a piece and a destination." end
    local white = mark == "X"
    local piece = pieceAt(game.board, from):lower()
    local promotion = PROMOTIONS[value % 4 + 1]
    local board = chessMove(game, from, to, white, promotion)
    if not board then return "Illegal move." end
    local notation = chessNotation(game, from, to, white, promotion)
    game.board = board
    game.enPassant = piece == "p" and math.abs(from - to) == 16 and (from + to) / 2 or nil
    if piece == "k" then game.rights = game.rights:gsub(white and "[KQ]" or "[kq]", "") end
    for square, right in pairs({[1] = "q", [8] = "k", [57] = "Q", [64] = "K"}) do
        if from == square or to == square then game.rights = game.rights:gsub(right, "") end
    end
    local check = inCheck(board, not white)
    if not hasChessMove(game, not white) then game.winner = check and mark or "draw" end
    if check then notation = notation .. (game.winner == mark and "#" or "+") end
    game.ply = game.ply + 1
    game.moves[#game.moves + 1] = notation
    if #game.moves > 64 then table.remove(game.moves, 1) end
end

local TRAIL_STOPS = {
    {300, "Kansas River", "river"}, {480, "Fort Kearny", "fort"},
    {830, "Platte River", "river"}, {1100, "Fort Laramie", "fort"},
    {1450, "Snake River", "river"}, {1690, "Fort Boise", "fort"}
}
local TRAIL_END = 2000
local TRAIL_TRAVEL = {"Travel steadily", "Push the pace", "Rest", "Forage"}
local TRAIL_RIVER = {"Ford the river", "Pay $8 for a ferry", "Wait for calmer water"}
local TRAIL_FORT = {"Continue west", "Buy 45 food for $10", "Repair wagon for $8", "Rest"}

local function newTrail()
    return {miles = 0, day = 1, food = 145, health = 100, wagon = 100, money = 34,
        nextStop = 1, selected = 1, event = "Your wagon leaves Independence. Oregon is 2,000 miles away."}
end

local function trailChoices(state)
    if state.stop and state.stop[3] == "river" then return TRAIL_RIVER end
    if state.stop and state.stop[3] == "fort" then return TRAIL_FORT end
    return TRAIL_TRAVEL
end

local function advanceTrail(state, miles)
    local stop = TRAIL_STOPS[state.nextStop]
    state.miles = math.min(TRAIL_END, state.miles + miles)
    if stop and state.miles >= stop[1] then
        state.miles, state.stop = stop[1], stop
        state.nextStop, state.selected, state.waited = state.nextStop + 1, 1, false
        state.event = state.event .. " You reached " .. stop[2] .. "."
    end
end

local function takeTrailTurn(state)
    local choice = state.selected
    local location = state.stop and state.stop[3]
    local foodCost, days = 0, 1
    if location == "river" then
        if choice == 3 then
            if state.waited then state.event = "The water is already calmer."; return end
            state.waited, foodCost, days = true, 5, 2
            state.event = "The river fell overnight. Fording is less risky now."
        elseif choice == 2 then
            if state.money < 8 then state.event = "You need $8 for the ferry."; return end
            state.money, state.stop, state.selected = state.money - 8, nil, 1
            foodCost = 5
            state.event = "The ferry carried the wagon across safely."
        else
            local mishap = math.random(100) <= (state.waited and 12 or 38)
            state.stop, state.selected, foodCost = nil, 1, 5
            if mishap then
                state.wagon, state.health, foodCost = state.wagon - 19, state.health - 12, 25
                state.event = "The wagon took on water. You lost food and damaged the wagon."
            else state.event = "You forded the river safely." end
        end
    elseif location == "fort" then
        if choice == 1 then
            state.stop, state.selected = nil, 1
            state.event = "You left the fort and returned to the trail."
            return
        elseif choice == 2 then
            if state.money < 10 then state.event = "You need $10 for food."; return end
            state.money, state.food, days = state.money - 10, state.food + 45, 0
            state.event = "You bought 45 food."
        elseif choice == 3 then
            if state.money < 8 then state.event = "You need $8 for repairs."; return end
            if state.wagon >= 100 then state.event = "The wagon is already in good repair."; return end
            state.money, state.wagon = state.money - 8, math.min(100, state.wagon + 40)
            state.event = "The fort's wheelwright repaired your wagon."
        else
            state.health, foodCost = math.min(100, state.health + 22), 6
            state.event = "Your party rested at the fort."
        end
    elseif choice == 3 then
        state.health, foodCost = math.min(100, state.health + 17), 7
        state.event = "A day of rest helped the party recover."
    elseif choice == 4 then
        local found = math.random(14, 34)
        state.food, foodCost = state.food + found, 5
        state.event = "You found " .. found .. " food while foraging."
    else
        local fast = choice == 2
        foodCost, days = fast and 14 or 10, 2
        state.health = state.health - (fast and 8 or 2)
        state.wagon = state.wagon - (fast and 7 or 2)
        local event = math.random(100)
        state.event = fast and "You pushed the wagon hard." or "You made steady progress."
        if event <= 10 then
            state.wagon = state.wagon - 10
            state.event = "A rocky stretch damaged the wagon."
        elseif event <= 20 then
            state.health = state.health - 10
            state.event = "Illness slowed the party."
        elseif event <= 30 then
            state.food = state.food + 12
            state.event = "Other travelers shared some food."
        end
        advanceTrail(state, fast and 145 or 100)
    end
    state.day = state.day + days
    if state.food < foodCost then
        state.health = state.health - 16
        state.event = state.event .. " The party went hungry."
    end
    state.food = math.max(0, state.food - foodCost)
    if state.health <= 0 then state.result = "The party could not continue."
    elseif state.wagon <= 0 then state.result = "The wagon broke down."
    elseif state.miles >= TRAIL_END then state.result = "You reached Oregon!"
    elseif state.day > 65 then state.result = "Winter closed the trail." end
end

local function gameAction(session, action, value, variant)
    if action == "game_snapshot" and not MULTIPLAYER_GAME_NAMES[variant] then
        return {supported = false}
    end
    variant = MULTIPLAYER_GAME_NAMES[variant] and variant or "tic"
    if action == "game_state" or action == "game_snapshot" then
        if value == gameRevision then
            if action == "game_snapshot" then return {variant = variant, revision = gameRevision} end
            return gameRevision
        end
        local list = {}
        for _, game in pairs(games) do
            if game.variant == variant then
                local open = not game.guest
                if game.party then open = not game.trail and #game.party < 4 end
                list[#list + 1] = {id = game.id, host = game.host.name, open = open,
                    players = game.party and #game.party or nil}
            end
        end
        table.sort(list, function(a, b) return a.id < b.id end)
        local game = session.game
        local current
        if game and game.variant == variant then
            current = {
                id = game.id, host = game.host.name, guest = game.guest and game.guest.name or "",
                mark = session == game.host and "X" or "O", board = game.board,
                turn = game.turn, winner = game.winner, check = game.variant == "chess" and inCheck(game.board, game.turn == "X"),
                moves = game.variant == "chess" and game.moves or nil, ply = game.ply,
                rematchRequested = session == game.rematchRequester,
                opponentRequested = game.rematchRequester ~= nil and session ~= game.rematchRequester
            }
            if variant == "pong" then current.pong = game.pong end
            if variant == "battleship" then
                current.target, current.fleet = battleshipBoards(game, current.mark)
                current.lastShot = game.lastShot
                current.lastShotHit = game.lastShotHit
                current.hits = game.hits[current.mark]
                current.sunk = game.sunk[current.mark]
                current.ready = fleetsReady(game)
                current.placed = #game.placements[current.mark]
                current.nextShip = FLEET_LENGTHS[current.placed + 1] or 0
            elseif variant == "trail" then
                current.trail = game.trail
                current.party = {}
                for i, member in ipairs(game.party) do current.party[i] = member.name end
                current.isHost = session == game.host
                current.canMove = game.party[game.trailTurn] == session
                current.controller = game.party[game.trailTurn] and game.party[game.trailTurn].name or ""
                current.votes, current.voted = 0, game.trailVotes[session] == true
                for _ in pairs(game.trailVotes) do current.votes = current.votes + 1 end
                if game.trail then
                    current.choices = trailChoices(game.trail)
                    local stop = game.trail.stop or TRAIL_STOPS[game.trail.nextStop]
                    current.destination = stop and stop[2] or "Oregon"
                end
            end
        end
        local notice = ""
        if session.gameDeparture and session.gameDeparture.variant == variant then
            notice = session.gameDeparture.message
            session.gameDeparture = nil
        end
        if action == "game_snapshot" then
            return {variant = variant, revision = gameRevision, matches = list, current = current, notice = notice}
        end
        return gameRevision, list, current or false, variant, notice
    end

    if action == "game_hint" then
        local game = session.game
        local mark = game and (session == game.host and "X" or "O")
        if not game or game.variant ~= "chess" or not game.guest or game.winner ~= "" or game.turn ~= mark
            or type(value) ~= "number" or value < 1 or value > 64 or value % 1 ~= 0 then return end
        local hints = {}
        for to = 1, 64 do
            if chessMove(game, value, to, mark == "X", "q") then hints[#hints + 1] = to end
        end
        return game.id, game.board, hints
    end

    if action == "game_create" then
        if session.game then return "Leave your current match first." end
        local count = 0
        for _ in pairs(games) do count = count + 1 end
        if count >= 16 then return "This room already has 16 matches." end
        gameSequence = gameSequence + 1
        local game = {id = gameSequence, host = session, variant = variant}
        resetGame(game)
        if variant == "trail" then game.party, game.trailTurn, game.trailVotes = {session}, 1, {} end
        games[game.id], session.game = game, game
        local announcement = session.name .. " opened a " .. MULTIPLAYER_GAME_NAMES[variant]
            .. " lobby (#" .. game.id .. ")."
        addHistory("Games", announcement, "[Games] " .. announcement)
    elseif action == "game_join" then
        if session.game then return "Leave your current match first." end
        local game = type(value) == "number" and games[value]
        if not game or game.variant ~= variant then return "That match is no longer open." end
        if variant == "trail" then
            if game.trail or #game.party >= 4 then return "That expedition is no longer open." end
            for _, member in ipairs(game.party) do
                if member.identity == session.identity then return "You already joined this expedition." end
            end
            game.party[#game.party + 1], session.game = session, game
        else
            if game.guest then return "That match is no longer open." end
            if game.host.identity == session.identity then return "A different player must join this match." end
            game.guest, session.game = session, game
        end
    elseif action == "game_leave" then
        if not session.game or session.game.variant ~= variant then return end
        closeGame(session)
        return "ok"
    elseif action == "game_rematch" then
        local game = session.game
        if game and game.variant == "trail" and variant == "trail" then
            if not game.trail or not game.trail.result then return "Finish the expedition first." end
            game.trailVotes[session] = true
            local all = true
            for _, member in ipairs(game.party) do if not game.trailVotes[member] then all = false; break end end
            if all then game.trail, game.trailTurn, game.trailVotes = newTrail(), 1, {} end
            gameRevision = gameRevision + 1
            return "ok"
        end
        if not game or game.variant ~= variant or game.winner == "" then return "Finish the match first." end
        if not game.guest then return "Finish the match first." end
        if game.rematchRequester == session then return "ok" end
        if game.rematchRequester then resetGame(game)
        else game.rematchRequester = session end
    elseif action == "game_move" then
        local game = session.game
        if not game or game.variant ~= variant then return "Join a match first." end
        if variant == "pong" then
            if not game.guest or game.winner ~= "" then return end
            local paddle = session == game.host and "left" or "right"
            if value == 1 or value == 2 then
                game.pong[paddle] = math.max(32, math.min(268, game.pong[paddle] + (value == 1 and -16 or 16)))
            elseif type(value) == "number" and value % 1 == 0 and value >= 132 and value <= 368 then
                game.pong[paddle] = value - 100
            else
                return
            end
            gameRevision = gameRevision + 1
            return "ok"
        end
        if variant == "trail" then
            if value == 10 and not game.trail and session == game.host then
                game.trail = newTrail()
            elseif not game.trail then return "Waiting for the host to start."
            elseif game.trail.result then return "The expedition has ended."
            elseif game.party[game.trailTurn] ~= session then return "Wait for your turn."
            elseif type(value) ~= "number" or value % 1 ~= 0 or value < 1 or value > #trailChoices(game.trail) then
                return "Choose a trail action."
            else
                game.trail.selected = value
                takeTrailTurn(game.trail)
                game.trailTurn = game.trailTurn % #game.party + 1
            end
            gameRevision = gameRevision + 1
            return "ok"
        end
        if game.winner ~= "" then return "This match has ended." end
        local mark = session == game.host and "X" or "O"
        if game.variant == "battleship" and type(value) == "number" and value >= 101 and value <= 301
            and value % 1 == 0 then
            local result = placeShip(game, mark, value)
            if result == "ok" then gameRevision = gameRevision + 1 end
            return result
        end
        if game.variant == "battleship" and not fleetsReady(game) then return "Both players must finish placing their ships." end
        if not game.guest then return "Waiting for another player." end
        if game.turn ~= mark then return "Wait for your turn." end
        if game.variant == "chess" then
            if type(value) ~= "number" or value < 256 or value >= 16640 or value % 1 ~= 0 then return "Invalid move." end
            local errorMessage = playChess(game, value, mark)
            if errorMessage then return errorMessage end
        elseif game.variant == "battleship" then
            if type(value) ~= "number" or value < 1 or value > 100 or value % 1 ~= 0 then
                return "Choose a square on the target grid."
            end
            if game.shots[mark][value] then return "You already fired at that square." end
            local opponent = mark == "X" and "O" or "X"
            local ship = game.fleets[opponent][value]
            game.shots[mark][value] = ship and "H" or "M"
            local coordinate = string.char(65 + (value - 1) % 10) .. (math.floor((value - 1) / 10) + 1)
            game.lastShot = session.name .. (ship and " hit " or " missed at ") .. coordinate
            game.lastShotHit = ship ~= nil
            if ship then
                ship.hits = ship.hits + 1
                game.hits[mark] = game.hits[mark] + 1
                if ship.hits == ship.length then
                    game.sunk[mark] = game.sunk[mark] + 1
                    game.lastShot = game.lastShot .. " and sunk " .. ship.name
                end
                if game.hits[mark] == FLEET_CELLS then game.winner = mark end
            end
        else
            if type(value) ~= "number" or value < 1 or value > 9 or value % 1 ~= 0
                or game.board:sub(value, value) ~= "." then return "Choose an empty square." end
            game.board = game.board:sub(1, value - 1) .. mark .. game.board:sub(value + 1)
            for _, line in ipairs(WIN_LINES) do
                if game.board:sub(line[1], line[1]) == mark and game.board:sub(line[2], line[2]) == mark
                    and game.board:sub(line[3], line[3]) == mark then
                    game.winner = mark
                    break
                end
            end
            if game.winner == "" and not game.board:find(".", 1, true) then game.winner = "draw" end
        end
        game.turn = mark == "X" and "O" or "X"
    else
        return
    end
    gameRevision = gameRevision + 1
    return "ok"
end

local function syncBoard()
    if not boardReady or not boardDirty or not owner():isValid() then return end
    local now = timer.curtime()
    local byPlayer = {}
    for sender, session in pairs(sessions) do
        if not session.removed and now < session.expires and sender:isValid() and session.owner:isValid() then
            local existing = byPlayer[session.owner]
            if not existing or session.id < existing.id then byPlayer[session.owner] = session end
        end
    end
    local members = {}
    for _, session in pairs(byPlayer) do
        members[#members + 1] = {id = session.id, name = session.name}
    end
    table.sort(members, function(a, b) return a.id < b.id end)

    local banned = {}
    for _, entry in pairs(bansById) do banned[#banned + 1] = entry end
    table.sort(banned, function(a, b) return a.id < b.id end)
    local banPages = math.max(1, math.ceil(#banned / MEMBERS_PER_PAGE))
    banPage = math.min(banPage, banPages)
    local banFirst = (banPage - 1) * MEMBERS_PER_PAGE + 1
    local banLast = math.min(#banned, banFirst + MEMBERS_PER_PAGE - 1)

    local bytes = 48 + #(roomName or "")
    for _, id in ipairs(channelOrder) do bytes = bytes + #channels[id].name + 5 end
    local count = math.min(#members, 255)
    for i = 1, count do bytes = bytes + #members[i].name + 5 end
    for i = banFirst, banLast do bytes = bytes + #banned[i].name + 5 end
    if net.getBytesLeft() < bytes then return end
    net.start("chatroom_board")
    net.writeString(roomName or "")
    net.writeUInt(count, 8)
    for i = 1, count do
        net.writeUInt(members[i].id, 32)
        net.writeString(members[i].name)
    end
    net.writeUInt(#banned, 32)
    net.writeUInt(banPage, 32)
    net.writeUInt(banPages, 32)
    net.writeUInt(banLast - banFirst + 1, 4)
    for i = banFirst, banLast do
        net.writeUInt(banned[i].id, 32)
        net.writeString(banned[i].name)
    end
    net.writeUInt(#channelOrder, 4)
    for _, id in ipairs(channelOrder) do
        net.writeUInt(id, 32)
        net.writeString(channels[id].name)
    end
    net.send(owner())
    boardDirty = false
end

local function syncPosts()
    if not boardReady or not postsDirty or not owner():isValid() then return end
    local posts = channels[postsChannel].history
    local bytes = 32
    for _, post in ipairs(posts) do bytes = bytes + #post.author + #post.message + 2 end
    if net.getBytesLeft() < bytes then return end
    net.start("chatroom_board_posts")
    net.writeUInt(postsChannel, 32)
    net.writeUInt(#posts, 5)
    for _, post in ipairs(posts) do
        net.writeString(post.author)
        net.writeString(post.message)
    end
    net.send(owner())
    postsDirty = false
end

local function saveRoom()
    local saved = setUserdata and getUserdata and pcall(function()
        local lines = {"MaxwellOSRoom2", tostring(owner():getSteamID64()), roomName, password}
        for _, id in ipairs(channelOrder) do
            if id ~= 1 then lines[#lines + 1] = "C\t" .. channels[id].name end
        end
        for identity, entry in pairs(bans) do
            lines[#lines + 1] = "B\t" .. identity .. "\t" .. entry.name
        end
        local record = table.concat(lines, "\n")
        setUserdata(record)
        assert(getUserdata() == record)
    end)
    if not saved then print("MaxwellOS: room settings could not be saved to this processor.") end
    return saved
end

do
    if getUserdata then
        local ok, raw = pcall(getUserdata)
        if ok and type(raw) == "string" and #raw <= 1048576 then
            local ownerId, name, secret, records
            local first = raw:find("\n", 1, true)
            local format = first and raw:sub(1, first - 1)
            if format == "MaxwellOSRoom1" or format == "MaxwellOSRoom2" then
                local second = raw:find("\n", first + 1, true)
                local third = second and raw:find("\n", second + 1, true)
                if third then
                    local fourth = raw:find("\n", third + 1, true)
                    if format == "MaxwellOSRoom2" or not fourth then
                        ownerId = raw:sub(first + 1, second - 1)
                        name = raw:sub(second + 1, third - 1)
                        secret = raw:sub(third + 1, fourth and fourth - 1 or #raw)
                        records = fourth and raw:sub(fourth + 1)
                    end
                end
            end
            if not ownerId and json and raw ~= "" then
                local decoded, saved = pcall(json.decode, raw)
                if decoded and type(saved) == "table" then
                    ownerId, name, secret = saved.owner, saved.name, saved.password
                end
            end
            if tostring(ownerId) == tostring(owner():getSteamID64())
                and type(name) == "string" and #name <= 28 and name ~= ""
                and not name:find("[^ -~]") and cleanChatText(name) == name and not containsSlurInName(name)
                and type(secret) == "string" and #secret <= 64 and not secret:find("[^ -~]") then
                roomName, password = name, secret
                local used = {general = true}
                for line in (records or ""):gmatch("[^\n]+") do
                    local channel = line:match("^C\t([ -~]+)$")
                    local identity, bannedName = line:match("^B\t([ -~]+)\t([A-Za-z0-9_-]+)$")
                    if channel and #channel <= 28 and #channelOrder < 12 and not used[channel:lower()]
                        and not containsSlurInName(channel) then
                        nextChannelId = nextChannelId + 1
                        channels[nextChannelId] = {name = channel, history = {}, sequence = 0}
                        channelOrder[#channelOrder + 1] = nextChannelId
                        used[channel:lower()] = true
                    elseif identity and #identity <= 64 and #bannedName <= 24 and not bans[identity] then
                        sessionSequence = sessionSequence + 1
                        local entry = {id = sessionSequence, identity = identity, name = bannedName}
                        bans[identity], bansById[entry.id] = entry, entry
                    end
                end
            end
        end
    end
end

    local name, secret = net.readString(), net.readString()
    if #name > 28 then return end
    name = cleanChatText(name)

        or #secret > 64 or secret:find("[^ -~]") then return end

            net.start("chatroom_setup_blocked")
            net.send(ply)
        end
        return
    end

    boardReady, boardDirty = true, true
    syncBoard()
end)

net.receive("chatroom_create_channel", function(_, ply)
    if ply ~= owner() or not roomName or net.getBytesLeft() < 128 then return end
    local id = net.readUInt(32)
    local value = net.readString()
    local name = #value <= 28 and cleanChatText(value) or ""
    local result
    if id == 1 then result = "General cannot be renamed."
    elseif id ~= 0 and not channels[id] then result = "That channel no longer exists."
    elseif name == "" or name:find("[^ -~]") then result = "Enter a channel name of up to 28 characters."
    elseif containsSlurInName(name) then result = "Channel name blocked. Choose another name."
    elseif id == 0 and #channelOrder >= 12 then result = "This room already has 12 channels."
    else
        for _, other in ipairs(channelOrder) do
            if other ~= id and channels[other].name:lower() == name:lower() then result = "That channel name is already in use."; break end
        end
    end
    local created = result == nil
    if created then
        if id == 0 then
            nextChannelId = nextChannelId + 1
            channels[nextChannelId] = {name = name, history = {}, sequence = 0}
            channelOrder[#channelOrder + 1] = nextChannelId
            result = "Channel created."
        else
            channels[id].name = name
            result = "Channel renamed."
        end
        if not saveRoom() then result = result .. " Save failed." end
        boardDirty = true
    end
    net.start("chatroom_channel_result")
    net.writeBool(created)
    net.writeString(result)
    net.send(ply)
end)

end)

local function removePlayerFromRoom(target, removal, now)
    for _, session in pairs(sessions) do
        if session.identity == target.identity then
            closeGame(session)
            session.removed = removal
            session.expires = now + 15
        end
    end
    local message = target.name .. " was " .. (removal == "banned" and "banned" or "kicked")
    addHistory("Moderation", message, "[Alert] " .. message, true)
    syncBoard()
end

net.receive("chatroom_moderate", function(_, ply)
    if ply ~= owner() then return end
    local action = net.readUInt(2)
    local id = net.readUInt(32)
    if action == 2 then
        local entry = bansById[id]
        if not entry then return end
        bans[entry.identity], bansById[id] = nil, nil
        saveRoom()
        boardDirty = true
        syncBoard()
        return
    end
    if action ~= 0 and action ~= 1 then return end

    local now = timer.curtime()
    local target
    for sender, session in pairs(sessions) do
        if session.id == id and not session.removed and now < session.expires and sender:isValid() then
            target = session
            break
        end
    end
    if not target or not target.owner:isValid() then return end
    local removal = action == 1 and "banned" or "kicked"
    if action == 1 then
        local entry = {id = id, identity = target.identity, name = target.name}
        bans[target.identity], bansById[id] = entry, entry
        saveRoom()
    end
    removePlayerFromRoom(target, removal, now)
end)

local PONG_STEP = 0.05
timer.create("chatroom_pong", PONG_STEP, 0, function()
    for _, game in pairs(games) do
        if game.pong and game.guest and game.winner == "" then
            local ball = game.pong
            local x, y = ball.x + ball.vx * PONG_STEP, ball.y + ball.vy * PONG_STEP
            if y <= 4 or y >= 296 then
                y = math.max(4, math.min(296, y))
                ball.vy = -ball.vy
            end
            local left = ball.vx < 0 and ball.x >= 22 and x <= 22
            local right = ball.vx > 0 and ball.x <= 418 and x >= 418
            if (left or right) and math.abs(y - (left and ball.left or ball.right)) <= 36 then
                x = left and 22 or 418
                ball.vx = (left and 1 or -1) * math.min(420, math.abs(ball.vx) + 18)
                ball.vy = (y - (left and ball.left or ball.right)) * 5
            end
            if x < 0 or x > 440 then
                local point = x < 0 and "b" or "a"
                ball[point] = ball[point] + 1
                if ball[point] >= 7 then game.winner = point == "a" and "X" or "O" end
                x, y = 220, 150
                ball.vx, ball.vy = point == "a" and 165 or -165, math.random(2) == 1 and 88 or -88
            end
            ball.x, ball.y = x, y
            gameRevision = gameRevision + 1
        end
    end
end)

hook.add("remote", "chatroom", function(sender, senderOwner, protocol, network, action, value, extra)
    if protocol ~= PROTOCOL or not roomName or type(action) ~= "string" then return end
    if action == "probe" then return roomName, NETWORK, password ~= "", CHIP_VERSION end
    if network ~= NETWORK then return end

    -- Discovery advertises only public room metadata.
    if action == "discover_index" then
        return chip():entIndex(), roomName, password ~= "", CHIP_VERSION
    end
    if action == "discover" then
        return chip(), roomName, password ~= "", CHIP_VERSION
    end

    if type(senderOwner) ~= "Player" or not senderOwner:isValid() then return end

    local now = timer.curtime()
    if action == "join" then
        local identity = playerIdentity(senderOwner)
        if bans[identity] then return "You are banned from this chatroom." end
        if now < (nextJoin[senderOwner] or 0) then
            return "Wait a second before trying again."
        end
        nextJoin[senderOwner] = now + 1
        if password ~= "" and (type(value) ~= "string" or #value > 64 or value ~= password) then
            return "Incorrect password."
        end
        local name = extra
        if name == nil then name = "Computer-" .. sender:entIndex() end
        if type(name) ~= "string" or #name > 24 or not name:match("^[A-Za-z0-9_-]+$") then
            return "Invalid computer name."
        end
        if containsSlurInName(name) then return "Computer name blocked. Choose another in Settings." end
        local nameKey = name:lower()
        if findComputer(nameKey, now, sender) then return "That computer name is already in use." end
        closeGame(sessions[sender])
        sessionSequence = sessionSequence + 1
        sessions[sender] = {
            owner = senderOwner, identity = identity, expires = now + 15, id = sessionSequence,
            name = name, nameKey = nameKey, privateMessages = {}, privateRevision = 0, channel = 1
        }
        boardDirty = true
        return "ok"
    end

    local session = sessions[sender]
    if not session or session.owner ~= senderOwner or now >= session.expires then return end
    if session.removed then
        if action == "read" then return session.removed end
        return
    end
    if action == "leave" then
        closeGame(session)
        sessions[sender] = nil
        boardDirty = true
        return
    end
    session.expires = now + 15
    if action == "rename" then
        if type(value) ~= "string" or #value > 24 or not value:match("^[A-Za-z0-9_-]+$") then
            return "Invalid computer name."
        end
        if containsSlurInName(value) then return "Computer name blocked. Choose another in Settings." end
        local nameKey = value:lower()
        if findComputer(nameKey, now, sender) then return "That computer name is already in use." end
        session.name, session.nameKey = value, nameKey
        boardDirty = true
        gameRevision = gameRevision + 1
        return "ok"
    end
    if action == "typing" then
        session.typingUntil = value == true and now + 4 or nil
        return
    end
    if action == "channel_directory" then
        local names = {}
        for i, id in ipairs(channelOrder) do names[i] = id .. ":" .. channels[id].name end
        return table.concat(names, "\n")
    end
    if action == "channel_join" and type(value) == "number" then
        if not channels[value] then return "That channel does not exist." end
        session.channel, session.typingUntil = value, nil
        return "ok"
    end
    if action == "private_state" then return privateState(session, now) end
    if action == "private_snapshot" then
        local state = privateState(session, now)
        local lines = {"PM1", tostring(state.revision), tostring(#state.recipients)}
        for _, entry in ipairs(state.recipients) do
            lines[#lines + 1] = entry.id .. ":" .. entry.name
        end
        for _, message in ipairs(state.messages) do lines[#lines + 1] = message end
        return table.concat(lines, "\n")
    end
    if action == "payment_list" then return paymentRecipients(senderOwner, now) end
    if action == "payment_target" and type(value) == "number" then
        local recipient, name, playerName = paymentRecipients(senderOwner, now, value)
        return recipient, name, playerName
    end
    if action == "payment_request" and type(value) == "number" then
        if type(extra) ~= "number" or extra < 1 or extra > 999999999 or extra % 1 ~= 0 then
            return "Enter a positive whole-dollar amount."
        end
        local recipient, _, _, processor = paymentRecipients(senderOwner, now, value)
        if not recipient then return "Recipient unavailable. No payment was requested." end
        if value == 0 then return requestPaymentFrom(senderOwner, extra) end
        local ok, results = pcall(hook.runRemote, processor, PROTOCOL, NETWORK, "payment_receive_request", senderOwner, extra)
        local reply = ok and results[1]
        return reply and type(reply[1]) == "string" and reply[1] or "The recipient's computer could not open a payment prompt."
    end
    if action == "trail_snapshot" and json then
        return json.encode(gameAction(session, "game_snapshot", value, "trail"))
    end
    if action:sub(1, 5) == "game_" then return gameAction(session, action, value, extra) end

    if action == "painting_latest" then
        local posts = channels[session.channel].history
        for i = #posts, 1, -1 do
            if posts[i].pixels then return posts[i].author, posts[i].pixels end
        end
        return "No paintings in this channel."
    end
    if action == "painting_post" then
        if now < (lastPost[senderOwner] or 0) then return "Wait a second before posting again." end
        if type(value) ~= "string" or #value ~= 2500 then return "Invalid painting." end
        for i = 1, #value do
            local byte = value:byte(i)
            if not ((byte >= 48 and byte <= 56) or (byte >= 65 and byte <= 90)) then return "Invalid painting." end
        end
        lastPost[senderOwner] = now + 1
        session.typingUntil = nil
        local thumbnail = {}
        for y = 3, 50, 6 do
            for x = 3, 50, 6 do
                local index = (y - 1) * 50 + x
                thumbnail[#thumbnail + 1] = value:sub(index, index)
            end
        end
        addHistory(session.name, "Shared a painting", session.name .. ": [Painting] F7 opens latest |"
            .. table.concat(thumbnail), nil, session.channel, value)
        return "Painting shared."
    end
    if action == "read" and type(value) == "number" then
        local channel = channels[session.channel]
        local unread, alerts = {}, {}
        for _, message in ipairs(channel.history) do
            if message.id > value then
                unread[#unread + 1] = message.line
                if message.alert then alerts[#unread] = true end
            end
        end
        local typists = {}
        for other, member in pairs(sessions) do
            if other ~= sender and not member.removed and member.channel == session.channel
                and member.expires > now and (member.typingUntil or 0) > now then
                typists[#typists + 1] = member.name
            end
        end
        table.sort(typists)
        local typing = typists[1] or ""
        if #typists > 1 then typing = typing .. " +" .. (#typists - 1) end
        return roomName, channel.sequence, unread, alerts, channel.name, session.channel, typing
    end

    if (action ~= "post" and action ~= "private_send") or type(value) ~= "string" then return end
    if action == "post" then session.typingUntil = nil end
    if #value > 160 then return "Use 1 to 160 bytes of text." end
    local message = cleanChatText(value)
    if message == "" then
        return "Use 1 to 160 bytes of text."
    end
    if containsSlur(message) then
        local identity = session.identity
        local strikes = (filterStrikes[identity] or 0) + 1
        if strikes >= 3 then
            filterStrikes[identity] = nil
            removePlayerFromRoom(session, "filter_kicked", now)
        else
            filterStrikes[identity] = strikes
        end
        return "your message was blocked"
    end

    if now < (lastPost[senderOwner] or 0) then
        return "Wait a second before posting again."
    end
    lastPost[senderOwner] = now + 1

    local name = session.name
    if action == "private_send" then
        for peer, recipient in pairs(sessions) do
            if recipient.id == extra and recipient.identity ~= session.identity and not recipient.removed
                and now < recipient.expires and peer:isValid() and recipient.owner:isValid() then
                appendPrivate(session, "To " .. recipient.name .. ": " .. message)
                appendPrivate(recipient, "From " .. name .. ": " .. message)
                return "ok"
            end
        end
        return "That computer is no longer connected."
    end
    addHistory(name, message, name .. ": " .. message, nil, session.channel)
    return "ok"
end)
return
end

local function wrapText(value, width, face)
    render.setFont(face)
    local lines, starts = {}, {}
    for offset in value:gmatch("()[^\128-\191][\128-\191]*") do
        starts[#starts + 1] = offset
    end
    local count = #starts
    starts[count + 1] = #value + 1
    local first = 1
    while first <= count do
        local low, high = first, count
        while low <= high do
            local middle = math.floor((low + high) / 2)
            local candidate = value:sub(starts[first], starts[middle + 1] - 1)
            if render.getTextSize(candidate) <= width then
                low = middle + 1
            else
                high = middle - 1
            end
        end
        local last = math.max(first, high)
        if last < count then
            for i = last, first + 1, -1 do
                if value:sub(starts[i], starts[i]) == " " then
                    last = i
                    break
                end
            end
        end
        lines[#lines + 1] = value:sub(starts[first], starts[last + 1] - 1)
        first = last + 1
    end
    return lines
end

local function preview(value, limit)
    local characters = {}
    for character in value:gmatch("[^\128-\191][\128-\191]*") do
        if #characters == limit then return table.concat(characters) .. "..." end
        characters[#characters + 1] = character
    end
    return table.concat(characters)
end

local function editSetupKey(key)
    local value = editingPassword and setupPassword or setupName
    local limit = editingPassword and 64 or 28
    if key == "Backspace" then
        value = value:sub(1, -2)
    elseif #value < limit then
        value = value .. key
    end
    if editingPassword then setupPassword = value else setupName = value end
    setupStatus = ""
end

local function useSetup(x, y)
    if setupWaiting and timer.curtime() < setupWaiting then return end
    if y >= 202 and y < 378 then
        local row = math.floor((y - 202) / 44) + 1
        local column = math.floor((x - 16) / 40) + 1
        local keys = (shifted and shiftedKeys or normalKeys)[row]
        if column <= #keys and (x - 16) % 40 < 36 and (y - 202) % 44 < 38 then
            editSetupKey(keys:sub(column, column))
        end
    elseif y >= 383 and y < 424 then
        if x < 130 then
            shifted = not shifted
        elseif x < 360 then
            editSetupKey(" ")
        else
            editSetupKey("Backspace")
        end
    elseif y >= 439 and y < 480 then
        if creatingChannel then
            if x < 150 then
                creatingChannel, setupWaiting, setupStatus = false, nil, ""
            elseif x >= 325 and setupName:find("%S") and net.getBytesLeft() >= 80 then
                setupWaiting = timer.curtime() + 5
                net.start("chatroom_create_channel")
                net.writeUInt(editingChannelId, 32)
                net.writeString(setupName)
                net.send()
            end
            return
        end
        if editingPassword and x < 150 then
            editingPassword, setupStatus = false, ""
        elseif x >= 325 then
            if not setupName:find("%S") then
                setupStatus = "Enter a room name first."
            elseif not editingPassword then
                editingPassword, shifted, setupStatus = true, false, ""
            elseif net.getBytesLeft() >= #setupName + #setupPassword + 32 then
                setupWaiting = timer.curtime() + 5
                setupStatus = "Starting room..."
                net.start("chatroom_setup")
                net.writeString(setupName)
                net.writeString(setupPassword)
                net.send()
            else
                setupStatus = "Connection busy. Try again shortly."
            end
        end
    end
end

local function postSlider()
    local viewport = POSTS_BOTTOM - POSTS_TOP
    local maximum = math.max(0, postHeight - viewport)
    local thumb = maximum > 0 and math.max(24, viewport * viewport / postHeight) or viewport
    return maximum, thumb, viewport - thumb
end

        end

    end)

    net.receive("chatroom_board", function()

        end
        members = {}
        for i = 1, net.readUInt(8) do
            local id, name = net.readUInt(32), net.readString()
            members[i] = {id = id, name = name}
        end
        bannedCount, bannedPage, bannedPages = net.readUInt(32), net.readUInt(32), net.readUInt(32)

        for i = 1, net.readUInt(4) do

        for _, entry in ipairs(tab == "banned" and banned or members) do
            if entry.id == selectedMember then selectionPresent = true end
        end
        if not selectionPresent then selectedMember = nil end

        if y >= 83 and y < 118 then

            selectedMember = nil
            confirmingChannel = nil

                    end

        end

                local index = channelIndex(postsChannel)

        if y >= 468 and y < 504 then

            local newPage = page
