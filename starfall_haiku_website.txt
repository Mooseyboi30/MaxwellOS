local poem, recentPoems = {}, {}
local sokka = false
local interval = 15
local nextPoemAt

local function buildLine(theme, target)
    local choices = {}
    for _, subject in ipairs(theme.subjects) do
        for _, action in ipairs(theme.actions) do
            for _, place in ipairs(theme.places) do
                if subject[2] + action[2] + place[2] == target then
                    choices[#choices + 1] = subject[1] .. " " .. action[1] .. " " .. place[1]
                end
            end
        end
    end
    local line = choices[math.random(#choices)]
    return line:sub(1, 1):upper() .. line:sub(2)
end

local function generate()
    nextPoemAt = timer.curtime() + interval
    local theme = themes[math.random(#themes)]
    sokka = math.random(10) == 1
    for attempt = 1, 12 do
        local candidate = {buildLine(theme, 5), buildLine(theme, 7), buildLine(theme, sokka and 6 or 5)}
        local key = table.concat(candidate, "\n")
        local repeated = false
        for _, previous in ipairs(recentPoems) do
            if previous == key then repeated = true; break end
        end
        if not repeated or attempt == 12 then
            poem = candidate
            recentPoems[#recentPoems + 1] = key
            if #recentPoems > 24 then table.remove(recentPoems, 1) end
            break
        end
    end
end

generate()
timer.create("haiku_auto_generate", 0.25, 0, function()
    if timer.curtime() >= nextPoemAt then generate() end
end)

local function frame()
    local seconds = math.max(0, math.ceil(nextPoemAt - timer.curtime()))
    return {
        {"rect", 0, 0, 512, 512, 19, 27, 42},

    }
end

if SERVER then
    local nextGenerate = 0
    hook.add("remote", "haiku_site", function(_, _, protocol)
        if protocol == "starfall_website_discover_v1" then
            return chip():entIndex(), NAME
        elseif protocol == "starfall_website_page_v1" then
            return chip():entIndex(), NAME, NAME, {"Press E on New Haiku."}, false, FONT, false, false, BUTTON
        elseif protocol == "starfall_website_screen_v1" then
            return frame()
        elseif protocol == "starfall_website_button_v1" and timer.curtime() >= nextGenerate then
            nextGenerate = timer.curtime() + 0.3
            generate()
        end
    end)
else
    local font = render.createFont(unpack(FONT))
    hook.add("StarfallUsed", "haiku_button", function(user, screen)
        if user ~= player() or not screen or not screen:isValid() or screen:getClass() ~= "starfall_screen" then return end
        local x, y = render.cursorPos(user, screen)
        if x and y and x >= 64 and x < 448 and y >= 360 and y < 424 then generate() end
    end)
    hook.add("render", "haiku_screen", function()
        render.setFont(font)
        for _, command in ipairs(frame()) do
            local kind, x, y, w, h, r, g, b, text = unpack(command)
            render.setColor(Color(r, g, b))
            if kind == "rect" then render.drawRect(x, y, w, h)
            else render.drawText(x, y, text) end
        end
    end)
