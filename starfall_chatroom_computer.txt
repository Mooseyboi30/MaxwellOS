local NETWORK = "lobby"
local PROTOCOL = "sf_chatroom_v2"

local function sendHistory()
    if not historyDirty or not ready or not owner():isValid() then return end
    local bytes = 32 + #(room or "") + #roomNotice + #channelName + #typingNames
    for _, entry in ipairs(history) do bytes = bytes + #entry.line + 2 end
    if net.getBytesLeft() < bytes then return end
    net.start("room_history")
    net.writeString(room or "")
    net.writeString(roomVersion)
    net.writeUInt(#history, 5)
    for _, entry in ipairs(history) do
        net.writeString(entry.line)
        net.writeBool(entry.alert)
    end
    net.writeString(roomNotice)
    net.writeUInt(sequence, 32)
    net.writeString(channelName)
    net.writeString(typingNames)
    net.send(owner())
    historyDirty = false
end

local function request(target, action, value, extra)
    if not target or not target:isValid() then return end
    -- A processor can still exist after its script stops or is reloaded.
    local ok, results = pcall(hook.runRemote, target, PROTOCOL, NETWORK, action, value, extra)
    if ok then return results[1] end
end

local function syncGames()
    if not watchingGames or not server or not room or net.getBytesLeft() < 1200 then return end
    local snapshot
    if watchedVariant == "trail" and json then
        local trailReply = request(server, "trail_snapshot", gameRevision)
        if trailReply and type(trailReply[1]) == "string" then
            local ok, decoded = pcall(json.decode, trailReply[1])
            if ok and type(decoded) == "table" then snapshot = decoded end
        end
    end
    if not snapshot then
        local reply = request(server, "game_snapshot", gameRevision, watchedVariant)
        snapshot = reply and reply[1]
    end
    local pollStatus = ""
    if type(snapshot) ~= "table" then
        local legacy = request(server, "game_state", gameRevision, watchedVariant)
        if legacy and type(legacy[1]) == "number"
            and (legacy[4] == watchedVariant or watchedVariant == "tic" or legacy[1] == gameRevision) then
            snapshot = {variant = watchedVariant, revision = legacy[1], matches = legacy[2],
                current = legacy[3], notice = legacy[5]}
        elseif legacy then
            pollStatus = "This server does not support this game."
        elseif watchedVariant == "trail" and tonumber(roomVersion) and tonumber(roomVersion) < 21.9 then
            pollStatus = "Update the room server to play Oregon Trail."
        else
            pollStatus = "No game response from the server. Retrying..."
        end
    end
    if type(snapshot) == "table" and (snapshot.supported == false or snapshot.variant ~= watchedVariant) then
        pollStatus = "This server does not support this game."
    end
    local revision = gameRevision
    if pollStatus == "" and type(snapshot.revision) == "number" then
        revision = snapshot.revision
        if revision ~= gameRevision and type(snapshot.matches) ~= "table" then
            revision = gameRevision
            pollStatus = "Incomplete game response from the server. Retrying..."
        end
    elseif pollStatus == "" then
        pollStatus = "Invalid game response from the server. Retrying..."
    end
    local changed = revision ~= gameRevision and type(snapshot.matches) == "table"
    if changed then hasWatchedMatch = type(snapshot.current) == "table" end
    if not gameVisible and not hasWatchedMatch then
        watchingGames = false
        timer.remove("chatroom_games_poll")
    end
    if not changed and gameNotice == "" and pollStatus == gamePollStatus then return end
    net.start("room_games_state")
    net.writeTable({
        changed = changed,
        matches = changed and snapshot.matches or {},
        current = changed and snapshot.current or nil,
        variant = watchedVariant,
        notice = changed and type(snapshot.notice) == "string" and snapshot.notice ~= "" and snapshot.notice
            or (gameNotice ~= "" and gameNotice or pollStatus),
        clearNotice = pollStatus == "" and gamePollStatus ~= ""
    })
    net.send(owner())
    if changed then gameRevision = revision end
    gameNotice, gamePollStatus = "", pollStatus
end

local function discover()
    local now = timer.curtime()
    if not ready or not owner():isValid() or now < nextDiscovery then return end
    nextDiscovery = now + 2
    local directory = {}
    local detail = "Check that the server processor is running."
    local function add(candidate, name, passwordRequired, version)
        if not candidate or not candidate:isValid() or candidate:getClass() ~= "starfall_processor"
            or type(name) ~= "string" or name == "" then return end
        directory[candidate:entIndex()] = {
            chip = candidate,
            name = name:gsub("%c", " "):sub(1, 40),
            passwordRequired = passwordRequired ~= false,
            version = type(version) == "string" and version:match("^%d%d?%d?%.%d$") or ""
        }
    end

    local function probe(candidate)
        local reply = request(candidate, "probe")
        if reply and type(reply[2]) == "string" then
            if reply[2] == NETWORK then
                add(candidate, reply[1], reply[3], reply[4])
            else
                detail = "Server network differs from " .. NETWORK:sub(1, 24) .. "."
            end
        else
            reply = request(candidate, "discover_index") or request(candidate, "discover")
            if reply then add(candidate, reply[2], reply[3], reply[4]) end
        end
    end

    if wiredServer and wiredServer:isValid() then
        detail = "Wired processor did not answer the chatroom probe."
        if wiredServer:getClass() == "starfall_processor" then probe(wiredServer) end
    end

    -- Probe actual processors so discovery also works when a broadcast has no replies.
    if find then
        local ok, processors = pcall(find.byClass, "starfall_processor")
        if ok then
            local self = chip()
            for _, candidate in ipairs(processors) do
                if candidate ~= self and candidate ~= wiredServer then probe(candidate) end
            end
        elseif not (wiredServer and wiredServer:isValid()) then
            detail = "Auto-scan unavailable. Wire RoomServer to ServerChip."
        end
    end

    local function collect(results)
        for _, result in ipairs(results) do
            local candidate, name = result[1], result[2]
            if type(candidate) == "number" then
                if candidate >= 1 and candidate <= 65535 and candidate % 1 == 0 then
                    candidate = entity(candidate)
                else
                    candidate = nil
                end
            elseif type(candidate) ~= "Entity" then
                candidate = nil
            end
            local existing = candidate and candidate:isValid() and directory[candidate:entIndex()]
            if not existing then add(candidate, name, result[3], result[4]) end
        end
    end
    local ok, results = pcall(hook.runRemote, nil, PROTOCOL, NETWORK, "discover_index")
    if ok then collect(results) end
    available = directory
    local ids = {}
    for id in pairs(directory) do ids[#ids + 1] = id end
    table.sort(ids)
    local signature = {}
    local count = math.min(#ids, 128)
    if count > 0 then detail = "" end
    local bytes = 32 + #detail
    for i = 1, count do
        local id = ids[i]
        local entry = directory[id]
        signature[i] = id .. ":" .. entry.version .. ":" .. tostring(entry.passwordRequired) .. ":" .. #entry.name .. ":" .. entry.name
        bytes = bytes + #entry.name + #entry.version + 5
    end
    signature = detail .. "\n" .. table.concat(signature)
    if signature == directorySignature or net.getBytesLeft() < bytes then return end
    net.start("room_directory")
    net.writeString(detail)
    net.writeUInt(count, 8)
    for i = 1, count do
        net.writeUInt(ids[i], 16)
        net.writeString(directory[ids[i]].name)
        net.writeBool(directory[ids[i]].passwordRequired)
        net.writeString(directory[ids[i]].version)
    end
    net.send(owner())
    directorySignature = signature
end

local function clearRoom(notice)
    server, room, sequence = nil, nil, 0
    roomVersion = ""
    channelName, typingNames = "General", ""
    roomNotice = notice or ""
    history = {}
    paymentQuote = nil
    watchingPrivate, privateSignature = false, nil
    watchingGames, gameVisible, hasWatchedMatch, gameRevision, gameNotice = false, false, false, -1, ""
    timer.remove("chatroom_games_poll")
    historyDirty = true
    nextDiscovery = 0
    sendHistory()
end

local function update()
    if not server then
        sendHistory()
        discover()
        return
    end

    local reply = request(server, "read", sequence)
    if reply and reply[1] == "filter_kicked" and reply[2] == nil then
        clearRoom("your message was blocked. You were kicked after 3 blocked messages.")
        return
    end
    if reply and reply[2] == nil and (reply[1] == "kicked" or reply[1] == "banned") then
        clearRoom("You were " .. reply[1] .. " from the chatroom.")
        return
    end
    local restarted = false
    if reply and type(reply[2]) == "number" and (reply[2] < sequence
        or (type(reply[5]) == "string" and reply[5] ~= channelName)) then
        sequence = 0
        history = {}
        restarted = true
        reply = request(server, "read", sequence)
    end
    if not reply or type(reply[1]) ~= "string" or type(reply[2]) ~= "number"
        or type(reply[3]) ~= "table" then
        clearRoom()
        return
    end

    local typing = type(reply[7]) == "string" and reply[7]:sub(1, 40) or ""
    historyDirty = historyDirty or restarted or room ~= reply[1] or sequence ~= reply[2] or typingNames ~= typing
    typingNames = typing
    room = reply[1]
    channelName = type(reply[5]) == "string" and reply[5] or "General"
    sequence = reply[2]
    local alerts = type(reply[4]) == "table" and reply[4] or {}
    for index, line in ipairs(reply[3]) do
        if type(line) == "string" then
            history[#history + 1] = {line = line, alert = alerts[index] == true}
            if #history > 24 then table.remove(history, 1) end
        end
    end
    sendHistory()
end

wire.adjustInputs({"KeyboardKeys", "KeyboardUser", "ServerChip", "Power"}, {"array", "entity", "entity", "number"})
wiredServer = wire.ports.ServerChip
local heldKeys = {}
local pendingKeys, firstPendingKey, lastPendingKey = {}, 1, 0
local keyRetryScheduled = false
local keyboardActive = false
local sentKeyboardActive, sentPower
local wirePowered = true

local function syncKeyboard()
    if not ready or net.getBytesLeft() < 32 then return end
    if keyboardActive == sentKeyboardActive and wirePowered == sentPower then return end
    net.start("room_keyboard")
    net.writeBool(keyboardActive)
    net.writeBool(wirePowered)
    net.send(owner())
    sentKeyboardActive, sentPower = keyboardActive, wirePowered
end

local function flushKeyboard()
    if not ready or not wirePowered or not keyboardActive or firstPendingKey > lastPendingKey then return end
    local count = math.min(lastPendingKey - firstPendingKey + 1, 32, math.floor(net.getBytesLeft()) - 32)
    if count > 0 then
        net.start("room_keys")
        net.writeUInt(count, 9)
        for i = firstPendingKey, firstPendingKey + count - 1 do
            net.writeUInt(pendingKeys[i], 8)
            pendingKeys[i] = nil
        end
        firstPendingKey = firstPendingKey + count
        if firstPendingKey > lastPendingKey then firstPendingKey, lastPendingKey = 1, 0 end
        net.send(owner())
    end
    if firstPendingKey <= lastPendingKey and not keyRetryScheduled then
        keyRetryScheduled = true
        timer.simple(0.025, function()
            keyRetryScheduled = false
            flushKeyboard()
        end)
    end
end

local function syncWirePower()
    local powered = not wire.isConnected(chip(), "Power") or wire.ports.Power ~= 0
    if powered ~= wirePowered then
        wirePowered = powered
        heldKeys = {}
        pendingKeys, firstPendingKey, lastPendingKey = {}, 1, 0
        if not powered then gameVisible = false end
    end
    syncKeyboard()
end

hook.add("input", "chatroom_keyboard", function(port, value)
    if port == "Power" then
        syncWirePower()
        return
    end
    if port == "ServerChip" then
        wiredServer = value
        nextDiscovery = 0
        return
    end
    if port == "KeyboardUser" then
        local active = value == owner()
        if active ~= keyboardActive then
            keyboardActive = active
            heldKeys = {}
            pendingKeys, firstPendingKey, lastPendingKey = {}, 1, 0
        end
        syncKeyboard()
        return
    end
    if port ~= "KeyboardKeys" or not wirePowered or not keyboardActive then return end

    local current = {}
    for _, key in ipairs(value) do
        if type(key) == "number" and key >= 0 and key <= 255 and key % 1 == 0
            and not current[key] then
            current[key] = true
            if ready and not heldKeys[key] and lastPendingKey - firstPendingKey < 127 then
                lastPendingKey = lastPendingKey + 1
                pendingKeys[lastPendingKey] = key
            end
        end
    end
    heldKeys = current
    flushKeyboard()
end)

hook.add("ClientInitialized", "chatroom_keyboard", function(ply)
    if ply ~= owner() then return end
    sentKeyboardActive = nil
    syncWirePower()
end)
timer.create("chatroom_keyboard_sync", 0.5, 0, syncWirePower)

local nextWebsiteRequest, requestedWebsiteId = 0, 0
net.receive("room_website_request", function(_, ply)
    if ply ~= owner() then return end
    local id = net.readUInt(16)
    local now = timer.curtime()
    if now < nextWebsiteRequest then return end
    nextWebsiteRequest = now + 0.4
    local sites, seen = {}, {}
    local function clean(value, limit)
        return type(value) == "string" and value:sub(1, limit):gsub("[^ -~]", " ") or ""
    end
    local function collect(results, target)
        if type(results) ~= "table" then return end
        for _, result in ipairs(results) do
            if type(result) == "table" and type(result[1]) == "number" then
                local candidate = target or (result[1] >= 1 and result[1] <= 65535
                    and result[1] % 1 == 0 and entity(result[1]))
                local name = clean(result[2], 32)
                if candidate and candidate:isValid() and candidate:getClass() == "starfall_processor" and name ~= "" then
                    local index = candidate:entIndex()
                    if not seen[index] then
                        local site = {id = index, name = name, chip = candidate}
                        seen[index] = site
                        sites[#sites + 1] = site
                    end
                end
            end
        end
    end
    local function discover(processor)
        local ok, results = pcall(hook.runRemote, processor, "starfall_website_discover_v1")
        if ok then collect(results, processor) end
    end
    if wiredServer and wiredServer:isValid() then discover(wiredServer) end
    if find then
        local ok, processors = pcall(find.byClass, "starfall_processor")
        if ok and type(processors) == "table" then
            local self = chip()
            for _, processor in ipairs(processors) do
                if processor ~= self and processor ~= wiredServer then discover(processor) end
            end
        end
    end
    discover(nil)
    table.sort(sites, function(first, second) return first.id < second.id end)

    local selected = seen[id]
    local page
    if selected then
        local ok, results = pcall(hook.runRemote, selected.chip, "starfall_website_page_v1")
        local result = ok and type(results) == "table" and results[1]
        if type(result) == "table" and type(result[4]) == "table" then
            local function bounded(value, lower, upper, default)
                if type(value) == "number" and value >= lower and value <= upper then return math.floor(value) end
                return default
            end
            local style = type(result[6]) == "table" and result[6] or {}
            local amounts = type(result[7]) == "table" and result[7] or {}
            page = {
                title = clean(result[3], 26), lines = {}, amounts = {},
                mode = result[5] == "demo_timeline_v1" and 1 or (result[5] == "donation_buttons_v1" and 2 or 0),
                face = clean(style[1], 40), size = bounded(style[2], 12, 64, 25),
                weight = bounded(style[3], 100, 1000, 500),

local function requestPaymentFrom(payer, amount)
    local api = economy and economy.requestMoney and economy or (darkrp and darkrp.requestMoney and darkrp)
    if not api then return "The in-game payment API is unavailable." end
    local ok, result = pcall(function()
        if api.canMakeMoneyRequest and not api.canMakeMoneyRequest(payer) then
            return "A payment request is pending or on cooldown."
        end
        if api.canAfford and not api.canAfford(payer, amount) then return "You cannot afford this payment." end
        api.requestMoney(payer, amount, "Chatroom payment", nil, nil, nil)
        return "ok"
    end)
    return ok and result or "Payment prompt failed. Check Starfall permissions."
end

net.receive("room_payment", function(_, ply)
    if ply ~= owner() or not room or not server or net.getBytesLeft() < 256 then return end
    local action, id, amount = net.readUInt(2), net.readUInt(32), net.readUInt(32)
    if action == 3 then
        paymentQuote = nil
        return
    end
    if action == 0 then
        local reply = request(server, "payment_list")
        if not reply or type(reply[1]) ~= "table" then
            paymentResult(ply, {notice = "This room server did not provide a payment list."})
            return
        end
        local recipients = {}
        local bytes = 128
        for i = 1, math.min(#reply[1], 128) do
            local entry = reply[1][i]
            if type(entry) == "table" and type(entry.id) == "number"
                and type(entry.name) == "string" and type(entry.playerName) == "string" then
                recipients[#recipients + 1] = {
                    id = entry.id, name = entry.name:sub(1, 24), playerName = entry.playerName:gsub("%c", " "):sub(1, 48)
                }
                local recipient = recipients[#recipients]
                bytes = bytes + #recipient.name + #recipient.playerName + 64
            end
        end
        if net.getBytesLeft() < bytes then
            paymentResult(ply, {notice = "Connection busy. Press R to refresh."})
            return
        end
        paymentResult(ply, {recipients = recipients})
        return
    end
    if action == 1 then
        paymentQuote = nil
        if amount < 1 or amount > 999999999 then
            paymentResult(ply, {notice = "Enter a positive whole-dollar amount."})
            return
        end
        local reply = request(server, "payment_target", id)
        local recipient = reply and reply[1]
        if type(recipient) ~= "Player" or not recipient:isValid() or recipient == ply then
            paymentResult(ply, {notice = "Recipient unavailable. Press R to refresh."})
            return
        end
        paymentSequence = paymentSequence + 1
        local name = recipient:getName():gsub("%c", " "):sub(1, 48)
        paymentQuote = {token = paymentSequence, id = id, recipient = recipient, amount = amount,
            expires = timer.curtime() + 20}
        paymentResult(ply, {quote = {token = paymentSequence, name = name, amount = amount}})
        return
    end
    if action ~= 2 then return end
    local quote = paymentQuote
    if not quote or quote.token ~= id then return end
    paymentQuote = nil
    if timer.curtime() >= quote.expires then
        paymentResult(ply, {finished = true, notice = "Confirmation expired. Review the payment again."})
        return
    end
    local reply = request(server, "payment_target", quote.id)
    if not reply or reply[1] ~= quote.recipient or not quote.recipient:isValid() then
        paymentResult(ply, {finished = true, notice = "Recipient unavailable. No payment was sent."})
        return
    end
    local response = request(server, "payment_request", quote.id, quote.amount)
    local result = response and response[1]
    local notice = "The server did not respond to the payment request."
    if result == "ok" then notice = "Payment prompt opened. Accept it in-game to pay."
    elseif type(result) == "string" then notice = result end
    paymentResult(ply, {finished = true, notice = notice})
end)

local function decodePrivateSnapshot(payload)
    if type(payload) ~= "string" or payload:sub(1, 4) ~= "PM1\n" then return end
    local lines = {}
    local first = 5
    while first <= #payload do
        local stop = payload:find("\n", first, true)
        lines[#lines + 1] = payload:sub(first, stop and stop - 1 or #payload)
        if not stop then break end
        first = stop + 1
    end
    local revision, count = tonumber(lines[1]), tonumber(lines[2])
    if not revision or not count or count < 0 or count > #lines - 2 or count % 1 ~= 0 then return end
    local snapshot = {revision = revision, recipients = {}, messages = {}}
    for i = 3, count + 2 do
        local line = lines[i]
        local colon = line:find(":", 1, true)
        local id = colon and tonumber(line:sub(1, colon - 1))
        if not id then return end
        snapshot.recipients[#snapshot.recipients + 1] = {id = id, name = line:sub(colon + 1)}
    end
    for i = count + 3, #lines do snapshot.messages[#snapshot.messages + 1] = lines[i] end
    return snapshot
end

local function syncPrivate()
    if not watchingPrivate or not room or net.getBytesLeft() < 256 then return end
    local reply = request(server, "private_snapshot")
    local snapshot = reply and decodePrivateSnapshot(reply[1])
    if not snapshot then
        local legacy = request(server, "private_state")
        if legacy and type(legacy[1]) == "table" then snapshot = legacy[1] end
    end
    local signature = {"unavailable"}
    local bytes = 128
    if type(snapshot) == "table" then
        signature = {tostring(snapshot.revision)}
        for _, entry in ipairs(snapshot.recipients) do
            signature[#signature + 1] = tostring(entry.id)
            bytes = bytes + #entry.name + 32
        end
        for _, message in ipairs(snapshot.messages) do bytes = bytes + #message + 12 end
    else
        local version = tonumber(roomVersion)
        snapshot = {notice = version and version < 19.3 and "This server predates private messaging."
            or "No private inbox reply. Retrying..."}
    end
    signature = table.concat(signature, ",")
    if signature == privateSignature or net.getBytesLeft() < bytes then return end
    net.start("room_private_state")
    net.writeTable(snapshot)
    net.send(owner())
    privateSignature = signature
end

local function isMentionNameByte(byte)
    return byte and ((byte >= 97 and byte <= 122) or (byte >= 48 and byte <= 57) or byte == 45 or byte == 95)
end

local function messageMentions(message, name)
    if name == "" then return false end
    local colon = message:find(": ", 1, true)
    if not colon then return false end
    local body = message:sub(colon + 2):lower()
    local wanted = name:lower()
    local from = 1
    while true do
        local at = body:find("@", from, true)
        if not at then return false end
        if (at == 1 or not isMentionNameByte(body:byte(at - 1)))
            and body:sub(at + 1, at + #wanted) == wanted
            and not isMentionNameByte(body:byte(at + #wanted + 1)) then
            return true
        end
        from = at + 1
    end
end

            settings.theme = math.floor(offset / THEME_STEP) + 1
        end
    elseif y >= 419 and y < 466 then
        applyTheme()
    end
end

local function wrapText(value, width, face)
    render.setFont(face)
    local lines, starts = {}, {}
    for offset in value:gmatch("()[^\128-\191][\128-\191]*") do
        starts[#starts + 1] = offset
    end
    local count = #starts
    starts[count + 1] = #value + 1
    local first = 1
    while first <= count do
        local low, high = first, count
        while low <= high do
            local middle = math.floor((low + high) / 2)
            local candidate = value:sub(starts[first], starts[middle + 1] - 1)
            if render.getTextSize(candidate) <= width then
                low = middle + 1
            else
                high = middle - 1
            end
        end
        local last = math.max(first, high)
        if last < count then
            for i = last, first + 1, -1 do
                if value:sub(starts[i], starts[i]) == " " then
                    last = i
                    break
                end
            end
        end
        lines[#lines + 1] = value:sub(starts[first], starts[last + 1] - 1)
        first = last + 1
    end
    return lines
end

end

PHONE.squadron = {}

function PHONE.squadron.wave()
    PHONE.squadron.enemies, PHONE.squadron.shots, PHONE.squadron.bolts = {}, {}, {}
    PHONE.squadron.time, PHONE.squadron.nextBolt, PHONE.squadron.nextDive = 0, 3.5, 4
    for row = 1, 3 do
        for column = 1, 8 do
            local x, y = 95 + (column - 1) * 46, 105 + row * 35
            PHONE.squadron.enemies[#PHONE.squadron.enemies + 1] = {
                x = -32, y = 84, baseX = x, baseY = y, kind = row,
                hp = row == 1 and 2 or 1,
                arrive = (row - 1) * 0.65 + (column - 1) * 0.08
            }
        end
    end
end

function PHONE.squadron.start()
    PHONE.squadron.x, PHONE.squadron.score, PHONE.squadron.lives, PHONE.squadron.level = 256, 0, 3, 1
    PHONE.squadron.wave()
end

function PHONE.squadron.key(key, enter)
    if not PHONE.squadron.lives or PHONE.squadron.lives == 0 then
        if enter then PHONE.squadron.start() end
    elseif key == 19 or key == 20 then
        PHONE.squadron.x = math.max(35, math.min(477, PHONE.squadron.x + (key == 19 and -18 or 18)))
    elseif (key == 32 or enter) and #PHONE.squadron.shots < 6 then
        PHONE.squadron.shots[#PHONE.squadron.shots + 1] = {x = PHONE.squadron.x, y = 411}
    end
end

function PHONE.squadron.update(now)
    local dt = math.min(0.05, math.max(0, now - (PHONE.squadron.lastTick or now)))
    PHONE.squadron.lastTick = now
    if not PHONE.squadron.lives or PHONE.squadron.lives == 0 then return end
    PHONE.squadron.time = PHONE.squadron.time + dt
    local enemies, shots, bolts = PHONE.squadron.enemies, PHONE.squadron.shots, PHONE.squadron.bolts
    for _, enemy in ipairs(enemies) do
        local t = PHONE.squadron.time
        local age = t - enemy.arrive
        local x = enemy.baseX + 44 * math.sin(t * 1.8)
        local y = enemy.baseY + 10 * math.sin(t * 2 + enemy.baseX / 55)
        enemy.active = age >= 0
        if age < 2 then
            local u = math.max(0, age / 2)
            local side = enemy.kind % 2 == 0 and 1 or -1
            local edge = 256 + side * 288
            enemy.x = edge + (x - edge) * u - side * 78 * math.sin(math.pi * u)
            enemy.y = 85 + (y - 85) * u + 125 * math.sin(math.pi * u)
        elseif enemy.dive then
            local u = (t - enemy.dive) / 2
            if u >= 1 then
                enemy.dive = nil
                enemy.x, enemy.y = x, y
            else
                enemy.x = x + 66 * math.sin(2 * math.pi * u)
                enemy.y = y + (393 - y) * math.sin(math.pi * u)
            end
        else
            enemy.x, enemy.y = x, y
        end
    end
    for i = #shots, 1, -1 do
        local shot = shots[i]
        shot.y = shot.y - 450 * dt
        for j = #enemies, 1, -1 do
            local enemy = enemies[j]
            if enemy.active and math.abs(shot.x - enemy.x) < 15 and math.abs(shot.y - enemy.y) < 15 then
                enemy.hp = enemy.hp - 1
                if enemy.hp == 0 then
                    table.remove(enemies, j)
                    PHONE.squadron.score = PHONE.squadron.score + (4 - enemy.kind) * 75
                end
                shot.y = 0
                break
            end
        end
        if shot.y < 90 then table.remove(shots, i) end
    end
    if #enemies == 0 then
        PHONE.squadron.level = PHONE.squadron.level + 1
        PHONE.squadron.wave()
        return
    end
    if PHONE.squadron.time >= PHONE.squadron.nextDive then
        local divers = {}
        for _, enemy in ipairs(enemies) do
            if enemy.kind ~= 1 and not enemy.dive then divers[#divers + 1] = enemy end
        end
        if #divers > 0 then divers[math.random(#divers)].dive = PHONE.squadron.time end
        PHONE.squadron.nextDive = PHONE.squadron.time + math.max(0.8, 2.8 - PHONE.squadron.level * 0.18)
    end
    if PHONE.squadron.time >= PHONE.squadron.nextBolt then
        local enemy = enemies[math.random(#enemies)]
        bolts[#bolts + 1] = {x = enemy.x, y = enemy.y + 10}
        PHONE.squadron.nextBolt = PHONE.squadron.time + math.max(0.28, 1.1 - PHONE.squadron.level * 0.09)
    end
    for i = #bolts, 1, -1 do
        local bolt = bolts[i]
        bolt.y = bolt.y + 200 * dt
        if math.abs(bolt.x - PHONE.squadron.x) < 15 and math.abs(bolt.y - 425) < 13 then
            PHONE.squadron.lives = PHONE.squadron.lives - 1
            PHONE.squadron.bolts = {}
            return
        end
        if bolt.y > 453 then table.remove(bolts, i) end
    end
end

local function calculatorInput(key)
    local state = calculator
    if key == "C" then
        state.entry, state.left, state.operator, state.error, state.result = "", nil, nil, nil, false
    elseif key == "Backspace" then
        state.entry, state.error, state.result = state.entry:sub(1, -2), nil, false
    elseif key == "." or (#key == 1 and key >= "0" and key <= "9") then
        if state.result or state.error then
            state.entry, state.error, state.result = "", nil, false
        end
        if #state.entry < 12 and (key ~= "." or not state.entry:find(".", 1, true)) then
            if key == "." and (state.entry == "" or state.entry == "-") then state.entry = state.entry .. "0" end
            state.entry = state.entry .. key
        end
    elseif key == "+" or key == "-" or key == "*" or key == "/" or key == "=" then
        local number = tonumber(state.entry)
        if not number then
            if key == "-" and state.entry == "" and not state.error then state.entry = "-"
            elseif state.operator and key ~= "=" then state.operator = key end
            return
        end
        if state.operator then
            if state.operator == "/" and number == 0 then
                state.entry, state.left, state.operator, state.error = "", nil, nil, "Cannot divide by zero."
                return
            end
            if state.operator == "+" then number = state.left + number
            elseif state.operator == "-" then number = state.left - number
            elseif state.operator == "*" then number = state.left * number
            else number = state.left / number end
        end
        if number ~= number or math.abs(number) == math.huge then
            state.entry, state.left, state.operator, state.error = "", nil, nil, "Result is out of range."
        elseif key == "=" then
            state.entry, state.left, state.operator, state.result = string.format("%.10g", number), nil, nil, true
        else
            state.left, state.operator, state.entry, state.result = number, key, "", false
        end
    end
end

local function calculatorKey(key, enter)
    if key == 172 then gameTab = "menu"; return end
    if enter then calculatorInput("=")
    elseif key == 127 or key == 8 then calculatorInput("Backspace")
    elseif key == 67 or key == 99 then calculatorInput("C")
    elseif key >= 128 and key <= 137 then calculatorInput(tostring(key - 128))
    elseif key >= 138 and key <= 141 then calculatorInput(({"/", "*", "-", "+"})[key - 137])
    elseif key == 143 then calculatorInput(".")
    elseif key >= 32 and key <= 126 then calculatorInput(string.char(key)) end
    return true
end

function PHONE.reportTyping(active)
    if room == "" or net.getBytesLeft() < 32 then return end
    local now = timer.curtime()
    if active and now < (PHONE.nextTyping or 0) then return end

local function versionStatus(version)
    local serverVersion = tonumber(version)
    if not serverVersion then return "Server version unknown", false end
    local computerVersion = tonumber(CHIP_VERSION)
    local text = "Server v" .. version
    if serverVersion < computerVersion then return text .. " (outdated)", true end
    if serverVersion > computerVersion then return text .. " (ahead)", true end
    return text, false
end

local function sendGameAction(command, value)
    if net.getBytesLeft() < 48 then return false end
    net.start("room_games_action")
    net.writeUInt(command, 3)
    net.writeUInt(value or 0, 32)
    net.send()
    return true
end

local function sendPrivateAction(command, recipient, message)
    if net.getBytesLeft() < #(message or "") + 48 then
        privateStatus = "Connection busy. Try again shortly."
        return false
    end
    net.start("room_private_action")
    net.writeUInt(command, 2)
    if command == 2 then
        net.writeUInt(recipient, 32)
        net.writeString(message)
    end
    net.send()
    return true
end

local function sendChannelAction(id)
    if net.getBytesLeft() < 48 then return end
    channelWaiting = timer.curtime() + 5
    net.start("room_channel_action")
    net.writeUInt(id, 32)
    net.send()
end

local function toggleChannels()
    if channelWaiting then return end
    showChannels = not showChannels
    if showChannels then
        channelStatus = "Loading channels..."
        sendChannelAction(0)
    end
end

local function handlePrivate(key, enter, character)
    if privateWaiting then return end
    if (key == 17 or key == 18) and #privateRecipients > 0 then
        privateSelection = math.max(1, math.min(#privateRecipients, privateSelection + (key == 17 and -1 or 1)))
    elseif key == 151 then
        privateScroll = math.min(privateScroll + 5, math.max(0, #privateRows - 6))
    elseif key == 152 then
        privateScroll = math.max(0, privateScroll - 5)
    elseif key == 127 or key == 8 then
        privateDraft = privateDraft:sub(1, -2)
    elseif enter and privateDraft:find("%S") then
        local recipient = privateRecipients[privateSelection]
        if not recipient then
            privateStatus = "Select a connected computer first."
        else
            privateWaiting = timer.curtime() + 5
            privateStatus = "Sending..."
            if not sendPrivateAction(2, recipient.id, privateDraft) then privateWaiting = nil end
        end
    elseif character and #privateDraft < 160 then
        privateDraft = privateDraft .. character
    end
end

local function sendPaymentAction(action, id, amount)
    if net.getBytesLeft() < 64 then
        paymentStatus = "Connection busy. Try again shortly."
        return false
    end
    net.start("room_payment")
    net.writeUInt(action, 2)
    net.writeUInt(id or 0, 32)
    net.writeUInt(amount or 0, 32)
    net.send()
    return true
end

local function refreshPayments()
    paymentStatus = "Loading recipients..."
    sendPaymentAction(0)
end

local function handlePayment(key, enter)
    if paymentWaiting then return end
    if paymentReview then
        if key == 127 or key == 8 then
            sendPaymentAction(3)
            paymentReview, paymentStatus = nil, ""
        elseif enter then
            local token = paymentReview.token
            paymentWaiting = timer.curtime() + 5
            paymentStatus = "Opening payment prompt..."
            if not sendPaymentAction(2, token) then paymentWaiting = nil end
        end
        return
    end
    if key == 17 then
        paymentSelection = math.max(1, paymentSelection - 1)
    elseif key == 18 then
        paymentSelection = math.max(1, math.min(#paymentRecipients, paymentSelection + 1))
    elseif key >= 48 and key <= 57 and #paymentAmount < 9 then
        paymentAmount = paymentAmount .. string.char(key)
    elseif key == 127 or key == 8 then
        paymentAmount = paymentAmount:sub(1, -2)
    elseif key == 82 or key == 114 then
        refreshPayments()
    elseif enter then
        local recipient = paymentRecipients[paymentSelection]
        local amount = tonumber(paymentAmount)
        if not recipient then
            paymentStatus = "Select a recipient first."
        elseif not amount or amount < 1 then
            paymentStatus = "Enter a positive whole-dollar amount."
        else
            paymentWaiting = timer.curtime() + 5
            paymentStatus = "Checking recipient..."
            if not sendPaymentAction(1, recipient.id, amount) then paymentWaiting = nil end
        end
    end
end

local function changeGameTab(tab)
    if room == "" and (MULTIPLAYER_TABS[tab] or tab == "payments") then return end
    if gameTab == "payments" and paymentWaiting then return end
    if MULTIPLAYER_TABS[tab] and tab ~= gameWatchTab
        and (currentGame or (gameEntryUntil and timer.curtime() < gameEntryUntil)) then return end
    if MULTIPLAYER_TABS[gameTab] and tab ~= gameTab and not sendGameAction(0) then return end
    if MULTIPLAYER_TABS[tab] then
        if not sendGameAction(1, MULTIPLAYER_TABS[tab]) then return end
        if tab ~= gameWatchTab then
            matches, currentGame, selectedMatch, gameStatus = {}, nil, 1, ""
            chessFrom, chessPromotion = nil, false
            chessHints, chessMoveScroll = {}, 0
            battleshipCursor, battleshipVertical = 1, false
        end
        gameWatchTab = tab
    end
    if showPayments then
        sendPaymentAction(3)
        showPayments, paymentReview = false, nil
    end
    if gameTab == "website" and tab ~= "website" then
        timer.remove("room_website_refresh")
        timer.remove("room_website_screen_refresh")
        websiteFrame, websiteFrameSite = nil, nil
    end
    gameTab = tab
    if tab == "website" then startWebsitePolling() end
    if tab ~= "menu" then lastApp = tab end
    if tab == "payments" then
        showPayments = true
        refreshPayments()
    end
end

local function newMinefield()
    return {
        cursor = 1, mines = {}, open = {}, flags = {}, opened = 0, flagCount = 0,
        count = MINE_DIFFICULTIES[mineDifficulty].count, noGuess = mineNoGuess
    }
end

local mineNeighbors = {}
local function neighbors(cell)
    if mineNeighbors[cell] then return mineNeighbors[cell] end
    local x = (cell - 1) % MINE_SIDE
    local y = math.floor((cell - 1) / MINE_SIDE)
    local cells = {}
    for row = math.max(0, y - 1), math.min(MINE_SIDE - 1, y + 1) do
        for column = math.max(0, x - 1), math.min(MINE_SIDE - 1, x + 1) do
            local nextCell = row * MINE_SIDE + column + 1
            if nextCell ~= cell then cells[#cells + 1] = nextCell end
        end
    end
    mineNeighbors[cell] = cells
    return cells
end

local function generateMines(first, count)
    local safe = {[first] = true}
    for _, neighbor in ipairs(neighbors(first)) do safe[neighbor] = true end
    local choices, mines = {}, {}
    for i = 1, MINE_CELLS do
        if not safe[i] then choices[#choices + 1] = i end
    end
    for i = 1, count do
        local index = math.random(i, #choices)
        choices[i], choices[index] = choices[index], choices[i]
        mines[choices[i]] = true
    end
    return mines
end

local function revealMineArea(field, cell)
    local queue, queued = {cell}, {[cell] = true}
    local index = 1
    while index <= #queue do
        local current = queue[index]
        local adjacent = neighbors(current)
        local count = 0
        for _, nextCell in ipairs(adjacent) do
            if field.mines[nextCell] then count = count + 1 end
        end
        field.open[current] = count
        field.opened = field.opened + 1
        if count == 0 then
            for _, nextCell in ipairs(adjacent) do
                if not queued[nextCell] and not field.open[nextCell] and not field.flags[nextCell] then
                    queued[nextCell] = true
                    queue[#queue + 1] = nextCell
                end
            end
        end
        index = index + 1
    end
end

local function mineConstraints(field)
    local constraints = {}
    for cell, count in pairs(field.open) do
        local cells, set, remaining = {}, {}, count
        for _, neighbor in ipairs(neighbors(cell)) do
            if field.flags[neighbor] then remaining = remaining - 1
            elseif not field.open[neighbor] then
                cells[#cells + 1] = neighbor
                set[neighbor] = true
            end
        end
        if #cells > 0 then constraints[#constraints + 1] = {cells = cells, set = set, remaining = remaining} end
    end
    return constraints
end

local function applyMineDeduction(field, cells, remaining)
    if remaining ~= 0 and remaining ~= #cells then return false end
    local changed = false
    for _, cell in ipairs(cells) do
        if not field.open[cell] and not field.flags[cell] then
            if remaining == 0 then
                revealMineArea(field, cell)
            else
                field.flags[cell] = true
            end
            changed = true
        end
    end
    return changed
end

local function solvesWithoutGuessing(mines, first, count, checkpoint)
    local simulation = {mines = mines, open = {}, flags = {}, opened = 0}
    revealMineArea(simulation, first)
    while simulation.opened < MINE_CELLS - count do
        if not checkpoint() then return false end
        local constraints = mineConstraints(simulation)
        local changed = false
        for _, rule in ipairs(constraints) do
            if applyMineDeduction(simulation, rule.cells, rule.remaining) then changed = true end
        end
        if not changed then
            local comparisons = 0
            for _, small in ipairs(constraints) do
                for _, large in ipairs(constraints) do
                    comparisons = comparisons + 1
                    if comparisons % 48 == 0 and not checkpoint() then return false end
                    if #small.cells < #large.cells then
                        local subset = true
                        for _, cell in ipairs(small.cells) do
                            if not large.set[cell] then subset = false; break end
                        end
                        if subset then
                            local difference = {}
                            for _, cell in ipairs(large.cells) do
                                if not small.set[cell] then difference[#difference + 1] = cell end
                            end
                            if applyMineDeduction(simulation, difference, large.remaining - small.remaining) then
                                changed = true
                                break
                            end
                        end
                    end
                end
                if changed then break end
            end
        end
        if not changed then return false end
    end
    return true
end

local function playMineCell(field, cell)
    if field.mines[cell] then
        playChatSound("explosion")
        field.result = "Mine hit. Press R to try again."
        return
    end
    revealMineArea(field, cell)
    if field.opened == MINE_CELLS - field.count then
        field.result = "You cleared the minefield! Press R to play again."
    end
end

local function revealMineCell()
    local field = minefield
    local cell = field.cursor
    if field.result or field.flags[cell] or field.open[cell] or field.generating then return end
    if field.started then
        playMineCell(field, cell)
    elseif not field.noGuess then
        field.mines, field.started = generateMines(cell, field.count), true
        playMineCell(field, cell)
    else
        if mineSearch then mineSearch() end
        field.generating, field.searchFailed = true, false
        mineSearch = coroutine.wrap(function()
            local function checkpoint()
                coroutine.yield(true)
                return minefield == field
            end
            for _ = 1, 120 do
                if minefield ~= field then return end
                local candidate = generateMines(cell, field.count)
                if solvesWithoutGuessing(candidate, cell, field.count, checkpoint) then
                    field.mines, field.started, field.generating = candidate, true, false
                    playMineCell(field, cell)
                    return
                end
                if not checkpoint() then return end
            end
            field.generating, field.searchFailed = false, true
        end)
    end
end

local function handleBattleship(key, enter)
    if key == 17 and battleshipCursor > 10 then battleshipCursor = battleshipCursor - 10
    elseif key == 18 and battleshipCursor <= 90 then battleshipCursor = battleshipCursor + 10
    elseif key == 19 and (battleshipCursor - 1) % 10 > 0 then battleshipCursor = battleshipCursor - 1
    elseif key == 20 and battleshipCursor % 10 > 0 then battleshipCursor = battleshipCursor + 1
    elseif currentGame.ready == false then
        if key == 82 or key == 114 then battleshipVertical = not battleshipVertical
        elseif key == 85 or key == 117 then sendGameAction(4, 301)
        elseif enter and currentGame.nextShip > 0 then
            sendGameAction(4, battleshipCursor + (battleshipVertical and 200 or 100))
        end
    elseif enter and currentGame.guest ~= "" and currentGame.winner == ""
        and currentGame.turn == currentGame.mark then sendGameAction(4, battleshipCursor) end
end

local function handleTrail(key, enter, character)
    if not currentGame.trail then
        if currentGame.isHost and character and character:lower() == "s" then sendGameAction(4, 10) end
        return
    end
    if currentGame.trail.result then
        if character and character:lower() == "r" then sendGameAction(6) end
        return
    end
    trailChoice = math.min(trailChoice, #currentGame.choices)
    if key == 17 then trailChoice = math.max(1, trailChoice - 1)
    elseif key == 18 then trailChoice = math.min(#currentGame.choices, trailChoice + 1)
    elseif enter and currentGame.canMove then sendGameAction(4, trailChoice) end
end

local function orientChessSquare(square, mark)
    return mark == "O" and 65 - square or square
end

local function moveChessCursor(key)
    local tile = orientChessSquare(chessCursor, currentGame.mark)
    if key == 17 and tile > 8 then tile = tile - 8
    elseif key == 18 and tile <= 56 then tile = tile + 8
    elseif key == 19 and (tile - 1) % 8 > 0 then tile = tile - 1
    elseif key == 20 and tile % 8 > 0 then tile = tile + 1 end
    chessCursor = orientChessSquare(tile, currentGame.mark)
end

local function submitChessMove(promotion)
    if sendGameAction(4, chessFrom * 256 + (chessCursor - 1) * 4 + promotion) then
        chessFrom, chessPromotion = nil, false
        chessHints = {}
    end
end

local function handleChess(key, enter, character)
    if chessPromotion then
        local promotion = character and ("qrbn"):find(character:lower(), 1, true)
        if promotion then submitChessMove(promotion - 1)
        elseif key == 8 or key == 127 then chessFrom, chessPromotion, chessHints = nil, false, {} end
        return
    end
    local moves = currentGame.moves or {}
    if key == 151 then
        chessMoveScroll = math.min(chessMoveScroll + 8, math.max(0, #moves - CHESS_MOVE_ROWS))
    elseif key == 152 then
        chessMoveScroll = math.max(0, chessMoveScroll - 8)
    end
    moveChessCursor(key)
    if not enter or currentGame.guest == "" or currentGame.turn ~= currentGame.mark
        or currentGame.winner ~= "" then return end
    local piece = currentGame.board:sub(chessCursor, chessCursor)
    local own = piece ~= "." and (piece == piece:upper()) == (currentGame.mark == "X")
    if own then
        chessFrom = chessFrom ~= chessCursor and chessCursor or nil
        chessHints = {}
        if chessFrom then sendGameAction(7, chessFrom) end
    elseif chessFrom then
        local moving = currentGame.board:sub(chessFrom, chessFrom):lower()
        if moving == "p" and (chessCursor <= 8 or chessCursor > 56) then chessPromotion = true
        else submitChessMove(0) end
    end
end

local function handleMines(key, enter)
    local field = minefield
    if not field then
        if key == 17 then mineDifficulty = math.max(1, mineDifficulty - 1)
        elseif key == 18 then mineDifficulty = math.min(#MINE_DIFFICULTIES, mineDifficulty + 1)
        elseif key == 32 then mineNoGuess = not mineNoGuess
        elseif enter then minefield = newMinefield() end
        return
    end
    local cell = field.cursor
    if key == 68 or key == 100 then
        minefield = nil
    elseif key == 82 or key == 114 then
        minefield = newMinefield()
    elseif field.generating then
        return
    elseif key == 17 and cell > MINE_SIDE then
        field.cursor = cell - MINE_SIDE
    elseif key == 18 and cell <= MINE_SIDE * (MINE_SIDE - 1) then
        field.cursor = cell + MINE_SIDE
    elseif key == 19 and (cell - 1) % MINE_SIDE > 0 then
        field.cursor = cell - 1
    elseif key == 20 and cell % MINE_SIDE > 0 then
        field.cursor = cell + 1
    elseif enter then
        revealMineCell()
    elseif (key == 70 or key == 102) and not field.result and not field.open[cell] then
        if field.flags[cell] then
            field.flags[cell] = nil
            field.flagCount = field.flagCount - 1
        elseif field.flagCount < field.count then
            field.flags[cell] = true
            field.flagCount = field.flagCount + 1
        end
    end
end

    local function handleGameKey(key, enter, character)
        if gameTab == "calculator" then calculatorKey(key, enter); return end
        if gameTab == "settings" then
            if key == 19 then settings.selectPage("color")
            elseif key == 20 then settings.selectPage("name")
            elseif settings.page == "color" then
                if key == 17 then settings.theme = math.max(1, settings.theme - 1)
                elseif key == 18 then settings.theme = math.min(#themes, settings.theme + 1)
                elseif enter then applyTheme() end
            elseif not settings.waiting then
                if enter then settings.saveName()
                elseif key == 8 or key == 127 then settings.name = settings.name:sub(1, -2)
                elseif character and #settings.name < 24 and character:match("^[A-Za-z0-9_-]$") then
                    settings.name = settings.name .. character
                end
            end
            return
        end
        if gameTab == "paint" then PHONE.paint.key(key, enter); return end
        if gameTab == "changelog" then
            if key == 19 or key == 151 then changelogPage = math.min(#CHANGELOG, changelogPage + 1)
            elseif key == 20 or key == 152 then changelogPage = math.max(1, changelogPage - 1)
            elseif key == 8 or key == 127 then changeGameTab("menu") end
            return
        end
        if gameTab == "website" then
            local page = websitePages[selectedWebsite]
            local scrollable = page and page.height and page.height > 512
            if key == 19 or (key == 17 and not scrollable) then stepWebsite(-1)
            elseif key == 20 or (key == 18 and not scrollable) then stepWebsite(1)
            elseif key == 17 then websiteControls.scroll(-64)
            elseif key == 18 then websiteControls.scroll(64)
            elseif key == 151 then websiteControls.scroll(-320)
            elseif key == 152 then websiteControls.scroll(320)
            elseif enter then websiteControls.buy()
            elseif key >= 49 and key <= 51 then requestWebsiteDonation(key - 48)
            elseif key == 8 or key == 127 then changeGameTab("menu") end
            return
        end
        if gameTab == "hud" then
            if enter then showChatHUD = not showChatHUD
            elseif key == 8 or key == 127 then changeGameTab("menu") end
            return
        end
        if gameTab == "payments" then
            if key == 168 then changeGameTab("menu")
            else handlePayment(key, enter) end
            return
        end
        if gameTab == "menu" then
            if selectedGame == 0 then
                if key == 19 then selectMenuTab(APPS)
                elseif key == 20 then selectMenuTab(GAMES)
                elseif key == 18 or enter then selectedGame = 1 end
            elseif key == 19 and selectedGame % 2 == 0 then selectedGame = selectedGame - 1
            elseif key == 20 and selectedGame % 2 == 1 and menuOptions[selectedGame + 1] then selectedGame = selectedGame + 1
            elseif key == 17 then selectedGame = math.max(0, selectedGame - 2)
            elseif key == 18 and menuOptions[selectedGame + 2] then selectedGame = selectedGame + 2
            elseif enter then enterSelectedGame() end
            return
        end
        if gameTab == "squadron" then PHONE.squadron.key(key, enter); return end
        if gameTab == "mines" then
            if key == 8 or key == 127 then changeGameTab("menu")
            else handleMines(key, enter) end
            return
        end
        if MULTIPLAYER_TABS[gameTab] then
            if currentGame then
                if (key == 8 or key == 127) and not chessPromotion then
                    sendGameAction(5)
                elseif character and character:lower() == "r" and currentGame.winner ~= "" then
                    sendGameAction(6)
                elseif gameTab == "chess" then
                    handleChess(key, enter, character)
                elseif gameTab == "battleship" then
                    handleBattleship(key, enter)
                elseif gameTab == "pong" then
                    if key == 17 then sendGameAction(4, 1)
                    elseif key == 18 then sendGameAction(4, 2) end
                elseif gameTab == "trail" then
                    handleTrail(key, enter, character)
                elseif key >= 49 and key <= 57 then
                    sendGameAction(4, key - 48)
                end
            elseif key == 8 or key == 127 then
                changeGameTab("menu")
            elseif key == 17 then
                selectedMatch = math.max(1, selectedMatch - 1)
            elseif key == 18 then
                selectedMatch = math.max(1, math.min(#matches, selectedMatch + 1))
            elseif character and character:lower() == "c" then
                if sendGameAction(2) then gameEntryUntil = timer.curtime() + 5 end
            elseif enter and matches[selectedMatch] then
                if sendGameAction(3, matches[selectedMatch].id) then gameEntryUntil = timer.curtime() + 5 end
            end
            return
        end
    end

    local function handleKey(key)
        local enter = key == 10 or key == 13 or key == 142
        if not themeConfigured then
            if key == 17 then
                settings.theme = math.max(1, settings.theme - 1)
            elseif key == 18 then
                settings.theme = math.min(#themes, settings.theme + 1)
            elseif enter then
                applyTheme()
            end
            return
        end
        local character
        if key >= 32 and key <= 126 then character = string.char(key) end
        if computerName == "" then
            if enter then
                if nameDraft ~= "" then
                    computerName, status = nameDraft, ""
                    settings.saveProfile()
                else
                    status = "Enter a computer name first."
                end
            elseif key == 127 or key == 8 then
                nameDraft = nameDraft:sub(1, -2)
            elseif character and character:match("^[A-Za-z0-9_-]$") and #nameDraft < 24 then
                nameDraft = nameDraft .. character
            end
            return
        end
        if key == 163 then
            toggleLastApp()
            return
        end
        if key == 168 and room ~= "" and not gameTab then
            if net.getBytesLeft() >= 32 then
                net.start("room_leave")
                net.send()
            end
            return
        end
        if not joining and not versionWarning and not showPrivate and not showChannels then
            if gameTab then handleGameKey(key, enter, character); return end
        end
        if room == "" then
            if joinUntil then return end
            if versionWarning then
                if key == 164 then
                    versionWarning = nil
                elseif enter then
                    local entry = directory[selected]
                    versionWarning = nil
                    joinSelected(entry)
                end
            elseif joining then
                if key == 164 then
                    joining, password, status = nil, "", ""
                elseif enter and password ~= "" then
                    submitJoin(joining, password)
                elseif key == 127 or key == 8 then
                    password = password:sub(1, -2)
                else
                    if character and #password < 64 then password = password .. character end
                end
            elseif key == 17 then
                selected = math.max(1, selected - 1)
                status = ""
            elseif key == 18 then
                selected = math.max(1, math.min(#directory, selected + 1))
                status = ""
            elseif enter and directory[selected] then
                local entry = directory[selected]
                local _, mismatch = versionStatus(entry.version)
                if mismatch then
                    versionWarning, status = entry, ""
                else
                    joinSelected(entry)
                end
            end
            return
        end
        if key == 164 and not gameTab and not showPayments and not showPrivate then
            toggleChannels()
            return
        end
        if showChannels then
            if channelWaiting then return end
            if key == 17 then selectedChannel = math.max(1, selectedChannel - 1)
            elseif key == 18 then selectedChannel = math.max(1, math.min(#channelNames, selectedChannel + 1))
            elseif key == 82 or key == 114 then sendChannelAction(0)
            elseif enter and channelIds[selectedChannel] then sendChannelAction(channelIds[selectedChannel]) end
            return
        end
        if key == 167 and not gameTab and not showPayments and not privateWaiting then
            local opening = not showPrivate
            if not opening or sendPrivateAction(1) then showPrivate = opening end
            return
        end
        if showPrivate then
            handlePrivate(key, enter, character)
            return
        end
        if key == 151 then
            scroll = math.min(scroll + 7, math.max(0, #rows - ROWS_VISIBLE))
        elseif key == 152 then
            scroll = math.max(0, scroll - 7)
        elseif waitingUntil then
            return
        elseif enter then
            if draft:find("%S") then
                if net.getBytesLeft() < #draft + 32 then
                    status = "Connection busy. Try again shortly."
                    return
                end
                waitingUntil = timer.curtime() + 5
                status = "Sending..."
                net.start("room_post")
                net.writeString(draft)
                net.send()
            end
        elseif key == 127 or key == 8 then
            local previous = draft
            draft = draft:sub(1, -2)
            if previous ~= draft then PHONE.reportTyping(draft ~= "") end
        else
            if character and #draft < 160 then
                draft = draft .. character
                PHONE.reportTyping(true)
            end
        end
    end

    function PHONE.setControlsLocked(enabled)
        if not enabled then
            -- Only undo a lock that this phone created, not another chip's lock.
            if PHONE.controlsLocked and hasPermission("input.lockcontrols") then
                pcall(input.lockControls, false)
            end
            PHONE.controlsLocked = false
            PHONE.lockNotice = ""
            return
        end
        if not PHONE.open or not poweredOn then return end
        if not hasPermission("input.lockcontrols") then
            PHONE.lockNotice = "Grant Lock Controls permission"
            PHONE.lockNoticeUntil = timer.curtime() + 4
            return
        end
        -- Starfall enforces a cooldown between locking attempts. Check before
        -- calling so a failed attempt does not throw an error in the chip.
        local canCheck, canLock = pcall(input.canLockControls)
        if not canCheck or not canLock then
            PHONE.lockNotice = "Relock cooldown (about 10 seconds)"
            PHONE.lockNoticeUntil = timer.curtime() + 3
            return
        end
        local ok = pcall(input.lockControls, true)
        if ok then
            PHONE.controlsLocked = true
            PHONE.lockNotice = ""
        else
            PHONE.lockNotice = "Unable to lock controls; try again"
            PHONE.lockNoticeUntil = timer.curtime() + 3
        end
    end
    function PHONE.setOpen(enabled)
        if not enabled then PHONE.setControlsLocked(false) end
        PHONE.open = enabled
        if enabled then PHONE.activateHud() end
        if hasPermission("input.enablecursor") then
            pcall(input.enableCursor, enabled)
        end
    end

    hook.add("inputpressed", "chatroom_phone_hotkey", function(button)
        if not hasPermission("input") then return end
        -- When typing is locked, Shift+P is an uppercase P, not the phone hotkey.
        -- Unlock with Alt or the on-screen button before toggling the phone.
        if button == KEY.P and input.isShiftDown() and not PHONE.controlsLocked then
            PHONE.setOpen(not PHONE.open)
            return
        end
        if not PHONE.open or not poweredOn or timer.curtime() < startupUntil then return end
        if button == KEY.ESCAPE then
            PHONE.setOpen(false)
            return
        end
        if button == KEY.LALT or button == KEY.RALT then
            if PHONE.controlsLocked then PHONE.setControlsLocked(false) end
            return
        end
        if button == KEY.F5 and room ~= "" then
            PHONE.openInbox()
            return
        end
        -- Only consume physical keyboard typing while game controls are locked.
        -- Otherwise the phone remains clickable while you play normally.
        if not PHONE.controlsLocked then return end
        local key = PHONE.keys[button]
        if key then
            local shifted = input.isShiftDown()
            if key >= 97 and key <= 122 and shifted then key = key - 32 end
            if shifted then key = PHONE.shiftKeys[key] or key end
            handleKey(key)
        end
    end)

    PHONE.mousePreviouslyDown = false
    hook.add("think", "chatroom_phone_mouse", function()
        -- Starfall always allows Alt to release its control lock. Reflect that
        -- emergency unlock in the button state before accepting more typing.
        if PHONE.controlsLocked then
            local ok, actual = pcall(input.isControlLocked)
            if ok and not actual then PHONE.controlsLocked = false end
        end
        if not PHONE.open or not hasPermission("input") or not hasPermission("input.enablecursor") then
            PHONE.mousePreviouslyDown = false
            return
        end
        local down = input.isMouseDown(MOUSE.LEFT)
        if down and not PHONE.mousePreviouslyDown then
            local mx, my = input.getCursorPos()
            local x, y, scale, face, fw = PHONE.geometry()
            if mx >= x + 14 and mx < x + fw - 14
                and my >= y + face + 82 and my < y + face + 112 then
                PHONE.setControlsLocked(not PHONE.controlsLocked)
            elseif mx >= x + fw - 43 and mx < x + fw - 17 and my >= y + 11 and my < y + 37 then
                PHONE.setOpen(false)
            elseif mx >= x + fw - 120 and mx < x + fw - 13
                and my >= y + face + 46 and my < y + face + 81 then
                PHONE.openInbox()
            elseif mx >= x + 14 and mx < x + 14 + face
                and my >= y + 46 and my < y + 46 + face then
                PHONE.computerClickAt((mx - x - 14) / scale, (my - y - 46) / scale)
            end
        end
        PHONE.mousePreviouslyDown = down
    end)

local websiteFonts, websiteFontCount = {}, 0
local function websiteFont(page)
    if page.face == "" then return smallFont end
    local size = math.max(8, math.floor(page.size * 0.625 + 0.5))
    local key = page.face .. "|" .. size .. "|" .. page.weight
    if websiteFonts[key] then return websiteFonts[key] end
    if websiteFontCount >= 16 then return smallFont end
    local ok, face = pcall(render.createFont, page.face, size, page.weight)
    websiteFontCount = websiteFontCount + 1
    websiteFonts[key] = ok and face or smallFont
    return websiteFonts[key]
end

local function drawWebsiteFrame(frame, face, scroll)
    scroll = scroll or 0
    local scale = 0.625
    local left, top = 96, 132
    for _, command in ipairs(frame) do
        if type(command) == "table" then
            local kind, x, y, w, h, r, g, b, value = unpack(command, 1, 9)
            if type(x) == "number" and type(y) == "number" and type(w) == "number" and type(h) == "number"
                and x >= 0 and y >= 0 and w >= 0 and h >= 0 and x + w <= 512 and y + h <= 1023
                and type(r) == "number" and r >= 0 and r <= 255
                and type(g) == "number" and g >= 0 and g <= 255
                and type(b) == "number" and b >= 0 and b <= 255 then
                y = y - scroll
                local color = Color(r, g, b)
                if kind == "rect" then
                    local first, last = math.max(0, y), math.min(512, y + h)
                    if last > first then
                        render.setColor(color)
                        render.drawRect(left + x * scale, top + first * scale, w * scale, (last - first) * scale)
                    end
                elseif kind == "text" and y >= 0 and y + h <= 512 and type(value) == "string" and #value <= 96 then
                    for _, line in ipairs(wrapText(value, w * scale, face)) do
                        label(left + x * scale, top + y * scale, line, color, face)
                        break
                    end
                end
            end
        end
    end
end

        for i, line in ipairs(wrapText(page.title, 396, websiteFont(page))) do
            label(58, 149 + (i - 1) * 21, line, Color(234, 245, 255), websiteFont(page))
        end
        for i, line in ipairs(page.lines) do
            for j, part in ipairs(wrapText(line, 396, websiteFont(page))) do
                label(58, 208 + (i - 1) * 38 + (j - 1) * 17, part, Color(198, 222, 239), websiteFont(page))
            end
        end
    end
    label(22, 460, "animations and other features might be laggy and or not work", alertColor, smallFont)
    if page and not page.title then label(64, 162, "Loading selected site...", muted, smallFont) end
    local hint = "Arrows switch sites  |  Backspace Apps"
    if page and page.height and page.height > 512 then
        render.setColor(panel)
        render.drawRect(426, 132, 54, 34)
        render.drawRect(426, 168, 54, 34)
        label(438, 140, "Up", accent, smallFont)
        label(434, 176, "Down", accent, smallFont)
        hint = "Up/Down scrolls  |  Left/Right switches sites"
    elseif page and page.donation then
        hint = "Press E on an amount or use 1-3."
    end
    if websiteControls.message ~= "" then hint = websiteControls.message end
    label(22, 483, hint, muted, smallFont)
end

local function renderChangelog()
    if updates.latest then
        label(22, 61, "Update: github.com/Mooseyboi30/MaxwellOS", accent, smallFont)
    end
    local entry = CHANGELOG[changelogPage]
    if renderedChangelogPage ~= changelogPage then
        changelogRows = wrapText(entry[2], 440, font)
        renderedChangelogPage = changelogPage
    end
    render.setColor(panel)
    render.drawRect(16, 86, 480, 310)
    label(32, 105, entry[1], accent, titleFont)
    for i, line in ipairs(changelogRows) do
        label(32, 161 + (i - 1) * 22, line, ink)
    end
    render.setColor(panel)
    render.drawRect(16, 414, 120, 40)
    render.drawRect(376, 414, 120, 40)
    label(30, 425, "< Older", changelogPage < #CHANGELOG and accent or muted)
    label(390, 425, "Newer >", changelogPage > 1 and accent or muted)
    label(220, 426, changelogPage .. " / " .. #CHANGELOG, muted, smallFont)
    label(22, 483, "Left/Right or PgUp/PgDn changes release  |  F1 Desktop", muted, smallFont)
end

local function renderAppsPanel()
    if MULTIPLAYER_TABS[gameTab] and activeAlert and timer.curtime() < alertUntil then
        label(22, 46, activeAlert, alertColor, smallFont)
    end
    if gameTab == "menu" then renderGameMenu()
    elseif gameTab == "calculator" then renderCalculator()
    elseif gameTab == "website" then renderWebsite()
    elseif gameTab == "paint" then PHONE.paint.draw()
    elseif gameTab == "changelog" then renderChangelog()
    elseif gameTab == "settings" then
        render.setColor(panel)
        render.drawRect(16, 72, 232, 26)
        render.drawRect(264, 72, 232, 26)
        label(28, 77, "Color", settings.page == "color" and accent or muted, smallFont)
        label(276, 77, "Computer name", settings.page == "name" and accent or muted, smallFont)
        if settings.page == "color" then
            renderThemeSetup()
        else
            label(22, 122, "Current name: " .. computerName, ink)
            render.setColor(panel)
            render.drawRect(16, 161, 480, 46)
            render.drawRect(16, 226, 160, 42)
            label(28, 174, settings.name .. "_", accent)
            label(28, 236, "Save name", accent)
            label(22, 285, settings.storageError and "Local profile could not be saved." or settings.message, muted, smallFont)
            label(22, 483, "Left/Right tabs  |  Type a name  |  Enter saves", muted, smallFont)
        end
    elseif gameTab == "hud" then
        label(22, 85, "Chat overlay", accent, titleFont)
        render.setColor(panel)
        render.drawRect(16, 134, 480, 59)
        label(28, 154, showChatHUD and "HUD on" or "HUD off", accent)
        label(22, 217, "Connect to the linked Starfall HUD to see room chat.", muted, smallFont)
        label(22, 244, "Press Enter or aim at the toggle and press E.", muted, smallFont)
        label(381, 438, "Back to Apps", accent, smallFont)
        label(22, 483, "Backspace returns to Apps.", muted, smallFont)
    elseif gameTab == "payments" then renderPayments()
    elseif gameTab == "squadron" then PHONE.squadron.draw()
    elseif gameTab == "mines" then renderMines()
    else renderMatchTab() end
end

function PHONE.drawComputerScreen()
    if not poweredOn then
        render.setColor(Color(9, 13, 19))
        render.drawRect(0, 0, 512, 512)
        label(22, 449, "Computer off", muted)
        label(22, 479, "Set the Wiremod Power input to 1.", muted, smallFont)
        return
    end
    local now = timer.curtime()
    if now < startupUntil then
        local progress = 1 - (startupUntil - now) / themes.startupDuration
        render.setColor(background)
        render.drawRect(0, 0, 512, 512)
        label(72, 200, "Starting computer", ink, titleFont)
        render.setColor(panel)
        render.drawRect(72, 260, 368, 10)
        if progress > 0 then
            render.setColor(accent)
            render.drawRect(72, 260, 368 * progress, 10)
        end
        return
    end
    render.clear(background)
    render.setColor(panel)
    render.drawRect(0, 0, 512, 72)
    if isOwner and not themeConfigured then
        renderThemeSetup()
        return
    end
    if isOwner and computerName == "" then
        label(22, 10, "Computer setup", ink, titleFont)
        label(22, 47, "Choose a computer name", muted, smallFont)
        label(22, 111, "Use 1–24 letters, numbers, hyphens or underscores.", ink, smallFont)
        render.setColor(panel)
        render.drawRect(16, 147, 480, 50)
        label(26, 163, nameDraft .. "_", accent)
        label(22, 483, status ~= "" and status or
            ((keyboardActive or PHONE.open) and "Type a name and press Enter." or "Look at the Wiremod Keyboard and press E."), muted, smallFont)
        return
    end
    if room == "" and not gameTab then
        render.drawRect(16, 423, 480, 48)
    elseif not gameTab and not showPayments and not showPrivate and not showChannels then
        render.drawRect(16, 372, 480, 102)
    end
    local title = room ~= "" and room:sub(1, 15) or "Chatroom"
    if gameTab == "menu" then title = "Desktop"
    elseif gameTab == "calculator" then title = "Calculator"
    elseif gameTab == "website" then title = "Website Viewer"
    elseif gameTab == "paint" then title = "Paint"
    elseif gameTab == "changelog" then title = "Changelog"
    elseif gameTab == "settings" then title = "Settings"
    elseif gameTab == "hud" then title = "HUD"
    elseif gameTab == "tic" then title = "Tic-Tac-Toe"
    elseif gameTab == "squadron" then title = "Star Squadron"
    elseif gameTab == "mines" then title = "Minesweeper"
    elseif gameTab == "chess" then title = "Chess"
    elseif gameTab == "battleship" then title = "Battleship"
    elseif gameTab == "pong" then title = "Pong"
    elseif gameTab == "trail" then title = "Oregon Trail"
    elseif showPayments then title = "Payments"
    elseif showPrivate then title = "Private messages"
    elseif showChannels then title = "Channels" end
    label(22, 10, title, ink, titleFont)

    if not isOwner then
        label(22, 46, "This terminal belongs to another player.", muted, smallFont)
        return
    end
    label(376, 46, updates.latest and ("Update v" .. updates.latest) or ("Computer v" .. CHIP_VERSION),
        updates.latest and alertColor or muted, smallFont)
    if room ~= "" then
        if not gameTab and not showPayments and not showPrivate then
            render.setColor(background)
            render.drawRect(265, 8, 100, 31)
            label(276, 15, "F2 Channels", accent, smallFont)
        end
        if showChannels then
            renderChannels()
            return
        end
        if showPrivate then
            renderPrivate()
            return
        end
    end
    if not joining and not versionWarning then
        render.setColor(background)
        render.drawRect(369, 8, 127, 31)
        local shortcut = gameTab == "menu" and "F1 Last app" or "F1 Desktop"
        label(380, 15, shortcut, accent, smallFont)
        if gameTab then
            renderAppsPanel()
            return
        end
    end
    local focused = keyboardActive or PHONE.open
    if room == "" then
        if versionWarning then
            local warning = versionStatus(versionWarning.version)
            label(22, 46, "Version mismatch", muted, smallFont)
            label(22, 90, versionWarning.name .. "  #" .. versionWarning.id, ink)
            label(22, 119, warning, accent, smallFont)
            label(22, 148, "Some features may behave differently.", muted, smallFont)
            label(22, 189, "Get matching versions:", ink, smallFont)
            label(22, 214, "https://github.com/Mooseyboi30/MaxwellOS", accent, smallFont)
            label(26, 439, "Enter continues  |  F2 goes back", ink)
            return
        end
        label(22, 46, joining and "Enter server password" or "Select a server", muted, smallFont)
        if joining then
            label(22, 90, joining.name .. "  #" .. joining.id, ink)
            local version = versionStatus(joining.version)
            label(22, 119, version, muted, smallFont)
            label(26, 439, "Password  " .. string.rep("*", math.min(32, #password)), accent)
        else
            if #directory == 0 then
                label(22, 90, directoryReceived and "No matching chatroom servers found." or "Looking for chatroom servers...", muted)
                if directoryReceived then
                    label(22, 119, directoryDetail, muted, smallFont)
                end
            end
            local first = math.max(1, selected - PICKER_ROWS + 1)
            for i = first, math.min(first + PICKER_ROWS - 1, #directory) do
                local entry = directory[i]
                local active = i == selected
                local y = 88 + (i - first) * 26
                if active then
                    render.setColor(panel)
                    render.drawRect(16, y - 3, 480, 24)
                end
                label(22, y, (active and "> " or "  ") .. entry.name:sub(1, 22) .. "  #" .. entry.id,
                    active and accent or ink)
                local version = versionStatus(entry.version)
                label(338, y, version, muted, smallFont)
            end
            label(26, 439, focused and "Up/Down selects   Enter opens" or "Use the wired keyboard to select", muted, smallFont)
        end
        local hint = joining and "Enter joins  |  F2 goes back  |  Alt exits" or "F1 Desktop  |  Alt exits"
        label(22, 483, status ~= "" and status or (focused and hint or "Look at the Wiremod Keyboard and press E"), muted, smallFont)
        return
    end
    if activeAlert and timer.curtime() < alertUntil then
        label(22, 46, activeAlert, alertColor, smallFont)
    else
        label(22, 46, "# " .. channelName .. "  |  " .. versionStatus(roomVersion), accent, smallFont)
    end

    if #rows == 0 then label(22, 91, "No messages yet.", muted) end
    local last = #rows - scroll
    local first = math.max(1, last - ROWS_VISIBLE + 1)
    for i = first, last do
        local row = rows[i]
        label(22, 88 + (i - first) * 22, row.text, row.alert and alertColor or (row.mention and accent or ink), row.face)
    end

    if PHONE.typingNames and PHONE.typingNames ~= "" then
        local names = PHONE.typingNames
        local firstName, others = names:match("^([^ ]+) %+(%d+)$")
        local who = firstName and (firstName .. " and " .. others .. " more are") or (names .. " is")
        local line = wrapText(who .. " typing in #" .. channelName, 468, smallFont)[1]
        label(22, 64, line, muted, smallFont)
    end
    local prompt = focused and ("> " .. draft .. "_")
        or (draft ~= "" and draft or "Use the wired keyboard to type")
    if prompt ~= lastPrompt then
        promptRows = wrapText(prompt, 330, emojiFont)
        lastPrompt = prompt
    end
    local firstPrompt = math.max(1, #promptRows - COMPOSER_ROWS + 1)
    for i = firstPrompt, #promptRows do
        label(26, 378 + (i - firstPrompt) * 18, promptRows[i], focused and accent or muted, emojiFont)
    end
    local hint = focused and "F1 Desktop  |  F2 Channels  |  F5 PM  |  F6 Leave"
        or "Look at the Wiremod Keyboard and press E"
    label(22, 483, status ~= "" and status or hint, muted, smallFont)
end

    if not isOwner or not PHONE.open then return end
    if not render.renderTargetExists("maxwellos_phone") then render.createRenderTarget("maxwellos_phone") end

    render.drawRect(x, y, fw, fh)
    render.setColor(panel)
    render.drawRect(x + 5, y + 5, fw - 10, fh - 10)
    render.setColor(accent)
    render.drawRect(x + 12, y + 41, face, 3)
    label(x + 17, y + 15, "POCKET COMPUTER", ink, smallFont)
    render.setColor(Color(115, 46, 49))
    render.drawRect(x + fw - 43, y + 11, 26, 26)
    label(x + fw - 35, y + 14, "X", ink, smallFont)

    render.setColor(Color(255, 255, 255))
    render.setRenderTargetTexture("maxwellos_phone")
    render.drawTexturedRectUV(x + 14, y + 46, face, face, 0, 0, 0.5, 0.5)
    render.setColor(accent)
    render.drawRect(x + fw / 2 - 26, y + fh - 9, 52, 3)

        label(x + 18, y + face + 64, PHONE.lockNotice, alertColor, smallFont)

            PHONE.controlsLocked and "Alt unlocks | X closes" or "Shift+P to close", muted, smallFont)
        label(x + fw - 108, y + face + 64,
            PHONE.unread > 0 and ("PM " .. PHONE.unread .. "  OPEN") or "PM  OPEN",

        PHONE.controlsLocked and "LOCK CONTROLS: ON" or "LOCK CONTROLS: OFF",
        PHONE.controlsLocked and accent or ink, smallFont)
end)
