-- Link a Starfall screen. Optionally wire the keyboard's Keys and User outputs
-- to KeyboardKeys and KeyboardUser. A wired Power value of 0 turns it off.

local SCREEN_MESSAGE = "demo_shared_screen"
local MAX_FRAME_BYTES = 12000
local CONTROL_MESSAGE = "demo_shared_control"
local CLICK, KEY_PRESS, KEEPALIVE, RELEASE = 0, 1, 2, 3

local function sendScreenFrame(data, recipient)
    if net.getBytesLeft() < #data + 64 then return false end
    net.start(SCREEN_MESSAGE)
    net.writeUInt(#data, 14)
    net.writeData(data, #data)
    net.send(recipient)
    return true
end

local function readScreenFrame(bits)
    local bytes = net.readUInt(14)
    if bytes > 0 and bytes <= MAX_FRAME_BYTES and bits >= bytes * 8 then
        return net.readData(bytes)
    end
end

if SERVER then
local pendingFrame
net.receive(SCREEN_MESSAGE, function(bits, sender)
    if sender ~= owner() then return end
    pendingFrame = readScreenFrame(bits) or pendingFrame
end)
timer.create("demo_screen_relay", 0.15, 0, function()
    if pendingFrame and sendScreenFrame(pendingFrame) then pendingFrame = nil end
end)

wire.adjustInputs({"KeyboardKeys", "KeyboardUser", "Power"}, {"array", "entity", "number"})
local heldKeys = {}
local pendingKeys, firstPendingKey, lastPendingKey = {}, 1, 0
local keyRetryScheduled = false
local keyboardUser
local sentPower
local wirePowered = true
local controller, controlUntil, nextKeyAt = nil, 0, 0

net.receive(CONTROL_MESSAGE, function(_, sender)
    local action = net.readUInt(2)
    local now = timer.curtime()
    if controller and (not controller:isValid() or now >= controlUntil) then controller = nil end
    if action == RELEASE then
        if sender == controller then controller = nil end
        return
    end
    if not wirePowered or (controller and controller ~= sender) or net.getBytesLeft() < 128 then return end
    if action == CLICK then
        local screen = entity(net.readUInt(16))
        local x, y = net.readUInt(9), net.readUInt(9)
        if not screen or not screen:isValid() or screen:getClass() ~= "starfall_screen"
            or sender:getEyeTrace().Entity ~= screen
            or sender:getEyePos():getDistance(screen:getPos()) > 200 then return end
        local linked = false
        for _, component in ipairs(chip():getLinkedComponents()) do
            if component == screen then linked = true; break end
        end
        if not linked then return end
        controller, controlUntil = sender, now + 5
        net.start(CONTROL_MESSAGE)
        net.writeUInt(CLICK, 2)
        net.writeUInt(x, 9)
        net.writeUInt(y, 9)
        net.send(owner())
    elseif sender ~= controller then
        return
    elseif action == KEY_PRESS then
        if now < nextKeyAt then return end
        nextKeyAt = now + 0.015
        local key = net.readUInt(8)
        net.start(CONTROL_MESSAGE)
        net.writeUInt(KEY_PRESS, 2)
        net.writeUInt(key, 8)
        net.send(owner())
        return
    end
    controlUntil = now + 5
    net.start(CONTROL_MESSAGE)
    net.writeUInt(KEEPALIVE, 2)
    net.send(sender)
end)

local function sendPower(ply)
    if net.getBytesLeft() < 32 then return false end
    net.start("demo_power")
    net.writeBool(wirePowered)
    net.send(ply)
    return true
end

local function flushKeyboard()
    if not wirePowered or not keyboardUser or not keyboardUser:isValid() or firstPendingKey > lastPendingKey then return end
    local count = math.min(lastPendingKey - firstPendingKey + 1, 32, math.floor(net.getBytesLeft()) - 32)
    if count > 0 then
        net.start("demo_keys")
        net.writeUInt(count, 9)
        for i = firstPendingKey, firstPendingKey + count - 1 do
            net.writeUInt(pendingKeys[i], 8)
            pendingKeys[i] = nil
        end
        firstPendingKey = firstPendingKey + count
        if firstPendingKey > lastPendingKey then firstPendingKey, lastPendingKey = 1, 0 end
        net.send(owner())
    end
    if firstPendingKey <= lastPendingKey and not keyRetryScheduled then
        keyRetryScheduled = true
        timer.simple(0.025, function()
            keyRetryScheduled = false
            flushKeyboard()
        end)
    end
end

local function syncWirePower()
    local powered = not wire.isConnected(chip(), "Power") or wire.ports.Power ~= 0
    if powered ~= wirePowered then
        wirePowered = powered
        controller = nil
        heldKeys = {}
        pendingKeys, firstPendingKey, lastPendingKey = {}, 1, 0
    end
    if wirePowered ~= sentPower and sendPower() then sentPower = wirePowered end
end

hook.add("input", "demo_keyboard", function(port, value)
    if port == "Power" then
        syncWirePower()
        return
    end
    if port == "KeyboardUser" then
        if value ~= keyboardUser then
            keyboardUser = value and value:isValid() and value:isPlayer() and value or nil
            heldKeys = {}
            pendingKeys, firstPendingKey, lastPendingKey = {}, 1, 0
        end
        return
    end
    if port ~= "KeyboardKeys" or not wirePowered or not keyboardUser then return end

    local current = {}
    for _, key in ipairs(value) do
        if type(key) == "number" and key >= 0 and key <= 255 and key % 1 == 0
            and not current[key] then
            current[key] = true
            if not heldKeys[key] and lastPendingKey - firstPendingKey < 127 then
                lastPendingKey = lastPendingKey + 1
                pendingKeys[lastPendingKey] = key
            end
        end
    end
    heldKeys = current
    flushKeyboard()
end)

hook.add("ClientInitialized", "demo_power", function(ply)
    syncWirePower()
    sendPower(ply)
end)
timer.create("demo_power_sync", 0.5, 0, syncWirePower)
local movementKeys = {17, 18, 19, 20, 32, 65, 68, 83, 87, 97, 100, 115, 119}
timer.create("demo_held_keys", 0.05, 0, function()
    if not wirePowered or not keyboardUser or not keyboardUser:isValid() or net.getBytesLeft() < 48 then return end
    local keys = {}
    for _, key in ipairs(movementKeys) do
        if heldKeys[key] then keys[#keys+1] = key end
    end
    if #keys == 0 then return end
    net.start("demo_held_keys")
    net.writeUInt(#keys, 4)
    for _, key in ipairs(keys) do net.writeUInt(key, 8) end
    net.send(owner())
end)
return
end

local fonts, fontIds = {}, {}

local drawMethods = {"drawRect", "drawRectOutline", "drawLine", "drawText"}
function render.createFont(...)
    local font = display.createFont(...)
    if not fontIds[font] then
        fonts[#fonts+1] = font
        fontIds[font] = #fonts
    end
    return font
end
function render.setFont(font)
    if frame then frame[#frame+1] = {5, fontIds[font]} end
    display.setFont(font)
end
function render.setColor(color)
    if frame then frame[#frame+1] = {6, color.r, color.g, color.b, color.a} end
    display.setColor(color)
end
for index, method in ipairs(drawMethods) do
    local draw = display[method]
    render[method] = function(...)
        if frame then frame[#frame+1] = {index, ...} end
        draw(...)
    end
end

function offline.player.stop()
    local app = offline.player
    if app.sound then pcall(function() app.sound:destroy() end) end
    app.sound, app.message = nil, nil
end

function offline.player.play()
    local app = offline.player
    local track = app.music[app.selected]
    app.stop()
    local ok = pcall(function()
        app.sound = sound.create(player(), "music/" .. track[2])
        app.sound:setSoundLevel(0)
        app.sound:play()
        app.sound:setVolume(0.5)
    end)
    if not ok then
        app.stop()
        app.message = "Playback unavailable. Check sound permission."
    else
        app.message = "Playing " .. track[1]
    end
end

function offline.player.move(delta)
    local app = offline.player
    app.selected = math.max(1, math.min(#app.music, app.selected + delta))
    app.top = math.max(app.selected - 3, math.min(app.top, app.selected))
end

function offline.player.click(x, y)
    local app = offline.player
    if y >= 400 and y < 446 then
        if x < 170 then app.stop()
        elseif x < 333 then app.move(-4)
        else app.move(4) end
    elseif y >= 94 and y < 394 and (y - 94) % 75 < 64 then
        local index = app.top + math.floor((y - 94) / 75)
        if app.music[index] then
            app.selected = index
            app.play()
        end
    end
end

offline.squadron = {}

function offline.squadron.wave()
    offline.squadron.enemies, offline.squadron.shots, offline.squadron.bolts = {}, {}, {}
    offline.squadron.time, offline.squadron.nextBolt, offline.squadron.nextDive = 0, 3.5, 4
    for row = 1, 3 do
        for column = 1, 8 do
            local x, y = 95 + (column - 1) * 46, 105 + row * 35
            offline.squadron.enemies[#offline.squadron.enemies + 1] = {
                x = -32, y = 84, baseX = x, baseY = y, kind = row,
                hp = row == 1 and 2 or 1,
                arrive = (row - 1) * 0.65 + (column - 1) * 0.08
            }
        end
    end
end

function offline.squadron.start()
    offline.squadron.x, offline.squadron.score, offline.squadron.lives, offline.squadron.level = 256, 0, 3, 1
    offline.squadron.wave()
end

function offline.squadron.key(key, enter)
    if not offline.squadron.lives or offline.squadron.lives == 0 then
        if enter then offline.squadron.start() end
    elseif key == 19 or key == 20 then
        offline.squadron.x = math.max(35, math.min(477, offline.squadron.x + (key == 19 and -18 or 18)))
    elseif (key == 32 or enter) and #offline.squadron.shots < 6 then
        offline.squadron.shots[#offline.squadron.shots + 1] = {x = offline.squadron.x, y = 411}
    end
end

function offline.squadron.update(now)
    local dt = math.min(0.05, math.max(0, now - (offline.squadron.lastTick or now)))
    offline.squadron.lastTick = now
    if not offline.squadron.lives or offline.squadron.lives == 0 then return end
    offline.squadron.time = offline.squadron.time + dt
    local enemies, shots, bolts = offline.squadron.enemies, offline.squadron.shots, offline.squadron.bolts
    for _, enemy in ipairs(enemies) do
        local t = offline.squadron.time
        local age = t - enemy.arrive
        local x = enemy.baseX + 44 * math.sin(t * 1.8)
        local y = enemy.baseY + 10 * math.sin(t * 2 + enemy.baseX / 55)
        enemy.active = age >= 0
        if age < 2 then
            local u = math.max(0, age / 2)
            local side = enemy.kind % 2 == 0 and 1 or -1
            local edge = 256 + side * 288
            enemy.x = edge + (x - edge) * u - side * 78 * math.sin(math.pi * u)
            enemy.y = 85 + (y - 85) * u + 125 * math.sin(math.pi * u)
        elseif enemy.dive then
            local u = (t - enemy.dive) / 2
            if u >= 1 then
                enemy.dive = nil
                enemy.x, enemy.y = x, y
            else
                enemy.x = x + 66 * math.sin(2 * math.pi * u)
                enemy.y = y + (393 - y) * math.sin(math.pi * u)
            end
        else
            enemy.x, enemy.y = x, y
        end
    end
    for i = #shots, 1, -1 do
        local shot = shots[i]
        shot.y = shot.y - 450 * dt
        for j = #enemies, 1, -1 do
            local enemy = enemies[j]
            if enemy.active and math.abs(shot.x - enemy.x) < 15 and math.abs(shot.y - enemy.y) < 15 then
                enemy.hp = enemy.hp - 1
                if enemy.hp == 0 then
                    table.remove(enemies, j)
                    offline.squadron.score = offline.squadron.score + (4 - enemy.kind) * 75
                end
                shot.y = 0
                break
            end
        end
        if shot.y < 90 then table.remove(shots, i) end
    end
    if #enemies == 0 then
        offline.squadron.level = offline.squadron.level + 1
        offline.squadron.wave()
        return
    end
    if offline.squadron.time >= offline.squadron.nextDive then
        local divers = {}
        for _, enemy in ipairs(enemies) do
            if enemy.kind ~= 1 and not enemy.dive then divers[#divers + 1] = enemy end
        end
        if #divers > 0 then divers[math.random(#divers)].dive = offline.squadron.time end
        offline.squadron.nextDive = offline.squadron.time + math.max(0.8, 2.8 - offline.squadron.level * 0.18)
    end
    if offline.squadron.time >= offline.squadron.nextBolt then
        local enemy = enemies[math.random(#enemies)]
        bolts[#bolts + 1] = {x = enemy.x, y = enemy.y + 10}
        offline.squadron.nextBolt = offline.squadron.time + math.max(0.28, 1.1 - offline.squadron.level * 0.09)
    end
    for i = #bolts, 1, -1 do
        local bolt = bolts[i]
        bolt.y = bolt.y + 200 * dt
        if math.abs(bolt.x - offline.squadron.x) < 15 and math.abs(bolt.y - 425) < 13 then
            offline.squadron.lives = offline.squadron.lives - 1
            offline.squadron.bolts = {}
            return
        end
        if bolt.y > 453 then table.remove(bolts, i) end
    end
end

end

function offline.paint.rebuildRow(y)
    local paint = offline.paint
    local row = paint.rows[y] or {}
    local runs = {}
    local previous
    for column = 1, 50 do
        local color = row[column]
        if color and color ~= previous then
            runs[#runs + 1] = {column, 1, color}
        elseif color then
            runs[#runs][2] = runs[#runs][2] + 1
        end
        previous = color
    end
    paint.runs[y] = runs
end

function offline.paint.pixel(x, y)
    local paint = offline.paint
    local row = paint.rows[y] or {}
    paint.rows[y] = row
    row[x] = paint.palette[paint.color] and paint.color or nil
    paint.rebuildRow(y)
end

function offline.paint.fill(x, y)
    local paint = offline.paint
    local target = (paint.rows[y] or {})[x]
    local replacement = paint.palette[paint.color] and paint.color or nil
    if target == replacement then return end

    local queue, changed = {}, {}
    local function visit(column, rowIndex)
        local row = paint.rows[rowIndex]
        if (row and row[column] or nil) ~= target then return end
        if not row then
            row = {}
            paint.rows[rowIndex] = row
        end
        row[column] = replacement
        changed[rowIndex] = true
        queue[#queue + 1] = (rowIndex - 1) * 50 + column - 1
    end

    visit(x, y)
    local head = 1
    while head <= #queue do
        local index = queue[head]
        local column, rowIndex = index % 50 + 1, math.floor(index / 50) + 1
        if column > 1 then visit(column - 1, rowIndex) end
        if column < 50 then visit(column + 1, rowIndex) end
        if rowIndex > 1 then visit(column, rowIndex - 1) end
        if rowIndex < 50 then visit(column, rowIndex + 1) end
        head = head + 1
    end
    for rowIndex in pairs(changed) do paint.rebuildRow(rowIndex) end
end

function offline.paint.apply()
    local paint = offline.paint
    if paint.bucket then paint.fill(paint.x, paint.y)
    else paint.pixel(paint.x, paint.y) end
end

function offline.paint.colorIndex(byte)
    if byte >= 65 and byte <= 90 then return byte - 56 end
    if byte >= 48 and byte <= 56 then return byte - 48 end
    return -1
end

function offline.paint.scrollPalette(delta)
    local paint = offline.paint
    paint.paletteTop = math.max(1, math.min(#paint.palette - 6, paint.paletteTop + delta))
end

function offline.paint.encode()
    local cells = {}
    for y = 1, 50 do
        local row = offline.paint.rows[y] or {}
        for x = 1, 50 do
            local index = row[x] or 0
            cells[#cells + 1] = string.char(index <= 8 and index + 48 or index + 56)
        end
    end
    return table.concat(cells)
end

function offline.paint.decode(data)
    if type(data) ~= "string" or #data ~= 2500 then return false end
    local rows, runs = {}, {}
    for y = 1, 50 do
        local row, spans, previous = {}, {}, 0
        for x = 1, 50 do
            local color = offline.paint.colorIndex(data:byte((y - 1) * 50 + x))
            if color < 0 or color > #offline.paint.palette then return false end
            if color > 0 then
                row[x] = color
                if color == previous then spans[#spans][2] = spans[#spans][2] + 1
                else spans[#spans + 1] = {x, 1, color} end
            end
            previous = color
        end
        rows[y], runs[y] = row, spans
    end
    return rows, runs
end

function offline.paint.action(action)
    local paint = offline.paint
    local path = paint.slot == 1 and "maxwellos_painting.txt" or ("maxwellos_painting_" .. paint.slot .. ".txt")
    if action == "slot" then
        paint.slot = paint.slot % 5 + 1
        paint.message = "Selected save slot " .. paint.slot .. "."
        return
    end
    if action == "save" then
        local ok = file and file.write and pcall(file.write, path, paint.encode())
        paint.message = ok and "Saved to slot " .. paint.slot .. "." or "Could not save painting."
    elseif action == "load" then
        local ok, data
        if file and file.read then ok, data = pcall(file.read, path) end
        local rows, runs = paint.decode(ok and data)
        if rows then paint.rows, paint.runs = rows, runs end
        paint.message = rows and "Loaded slot " .. paint.slot .. "." or ("No readable painting in slot " .. paint.slot .. ".")
    end
end

calculator.functions = {
    sqrt = math.sqrt, ["x^2"] = function(x) return x * x end,
    log = function(x) return math.log(x) / math.log(10) end, ln = math.log,
    ["%"] = function(x) return x / 100 end, ["+/-"] = function(x) return -x end,
    ["1/x"] = function(x) return 1 / x end,
    sin = math.sin, cos = math.cos, tan = math.tan
}

local function calculatorInput(key)
    local state = calculator
    if state.error and key == "=" then return end
    if key == "C" then
        state.entry, state.left, state.operator, state.error, state.result = "", nil, nil, nil, false
    elseif key == "Backspace" then
        state.entry, state.error, state.result = state.entry:sub(1, -2), nil, false
    elseif key == "." or (#key == 1 and key >= "0" and key <= "9") then
        if state.result or state.error then
            state.entry, state.error, state.result = "", nil, false
        end
        if #state.entry < 12 and (key ~= "." or not state.entry:find(".", 1, true)) then
            if key == "." and (state.entry == "" or state.entry == "-") then state.entry = state.entry .. "0" end
            state.entry = state.entry .. key
        end
    elseif key == "Angle" then
        state.radians = not state.radians
    elseif key == "pi" or key == "e" then
        state.entry = string.format("%.15g", key == "pi" and math.pi or math.exp(1))
        state.error, state.result = nil, not state.operator
    elseif state.functions[key] then
        local number = tonumber(state.entry)
        if not number then return end
        if key == "sin" or key == "cos" or key == "tan" then
            if not state.radians then number = number * math.pi / 180 end
            if key == "tan" and math.abs(math.cos(number)) < 1e-12 then
                state.error = "Tangent is undefined."
                return
            end
        end
        local result = state.functions[key](number)
        if result ~= result or math.abs(result) == math.huge then
            state.error = "Undefined or out of range."
        else
            state.entry = string.format("%.15g", result)
            state.error, state.result = nil, true
        end
    elseif key == "+" or key == "-" or key == "*" or key == "/" or key == "^" or key == "=" then
        local number = tonumber(state.entry)
        if not number then
            if key == "-" and state.entry == "" and not state.error then state.entry = "-"
            elseif state.operator and key ~= "=" then state.operator = key end
            return
        end
        if state.operator then
            if state.operator == "/" and number == 0 then
                state.entry, state.left, state.operator, state.error, state.result = "", nil, nil, nil, false
                gameTab = "menu"
                startupUntil = timer.curtime() + 2.5 + STARTUP_DURATION
                return
            end
            if state.operator == "+" then number = state.left + number
            elseif state.operator == "-" then number = state.left - number
            elseif state.operator == "*" then number = state.left * number
            elseif state.operator == "^" then number = state.left ^ number
            else number = state.left / number end
        end
        if number ~= number or math.abs(number) == math.huge then
            state.entry, state.left, state.operator, state.error = "", nil, nil, "Result is out of range."
        elseif key == "=" then
            state.entry, state.left, state.operator, state.result = string.format("%.10g", number), nil, nil, true
        else
            state.left, state.operator, state.entry, state.result = number, key, "", false
        end
    end
end

local function newMinefield()
    return {
        cursor = 1, mines = {}, open = {}, flags = {}, opened = 0, flagCount = 0,
        count = MINE_DIFFICULTIES[mineDifficulty].count, noGuess = mineNoGuess
    }
end

local mineNeighbors = {}
local function neighbors(cell)
    if mineNeighbors[cell] then return mineNeighbors[cell] end
    local x = (cell - 1) % MINE_SIDE
    local y = math.floor((cell - 1) / MINE_SIDE)
    local cells = {}
    for row = math.max(0, y - 1), math.min(MINE_SIDE - 1, y + 1) do
        for column = math.max(0, x - 1), math.min(MINE_SIDE - 1, x + 1) do
            local nextCell = row * MINE_SIDE + column + 1
            if nextCell ~= cell then cells[#cells + 1] = nextCell end
        end
    end
    mineNeighbors[cell] = cells
    return cells
end

local function generateMines(first, count)
    local safe = {[first] = true}
    for _, neighbor in ipairs(neighbors(first)) do safe[neighbor] = true end
    local choices, mines = {}, {}
    for i = 1, MINE_CELLS do
        if not safe[i] then choices[#choices + 1] = i end
    end
    for i = 1, count do
        local index = math.random(i, #choices)
        choices[i], choices[index] = choices[index], choices[i]
        mines[choices[i]] = true
    end
    return mines
end

local function revealMineArea(field, cell)
    local queue, queued = {cell}, {[cell] = true}
    local index = 1
    while index <= #queue do
        local current = queue[index]
        local adjacent = neighbors(current)
        local count = 0
        for _, nextCell in ipairs(adjacent) do
            if field.mines[nextCell] then count = count + 1 end
        end
        field.open[current] = count
        field.opened = field.opened + 1
        if count == 0 then
            for _, nextCell in ipairs(adjacent) do
                if not queued[nextCell] and not field.open[nextCell] and not field.flags[nextCell] then
                    queued[nextCell] = true
                    queue[#queue + 1] = nextCell
                end
            end
        end
        index = index + 1
    end
end

local function mineConstraints(field)
    local constraints = {}
    for cell, count in pairs(field.open) do
        local cells, set, remaining = {}, {}, count
        for _, neighbor in ipairs(neighbors(cell)) do
            if field.flags[neighbor] then remaining = remaining - 1
            elseif not field.open[neighbor] then
                cells[#cells + 1] = neighbor
                set[neighbor] = true
            end
        end
        if #cells > 0 then constraints[#constraints + 1] = {cells = cells, set = set, remaining = remaining} end
    end
    return constraints
end

local function applyMineDeduction(field, cells, remaining)
    if remaining ~= 0 and remaining ~= #cells then return false end
    local changed = false
    for _, cell in ipairs(cells) do
        if not field.open[cell] and not field.flags[cell] then
            if remaining == 0 then
                revealMineArea(field, cell)
            else
                field.flags[cell] = true
            end
            changed = true
        end
    end
    return changed
end

local function solvesWithoutGuessing(mines, first, count, checkpoint)
    local simulation = {mines = mines, open = {}, flags = {}, opened = 0}
    revealMineArea(simulation, first)
    while simulation.opened < MINE_CELLS - count do
        if not checkpoint() then return false end
        local constraints = mineConstraints(simulation)
        local changed = false
        for _, rule in ipairs(constraints) do
            if applyMineDeduction(simulation, rule.cells, rule.remaining) then changed = true end
        end
        if not changed then
            local comparisons = 0
            for _, small in ipairs(constraints) do
                for _, large in ipairs(constraints) do
                    comparisons = comparisons + 1
                    if comparisons % 48 == 0 and not checkpoint() then return false end
                    if #small.cells < #large.cells then
                        local subset = true
                        for _, cell in ipairs(small.cells) do
                            if not large.set[cell] then subset = false; break end
                        end
                        if subset then
                            local difference = {}
                            for _, cell in ipairs(large.cells) do
                                if not small.set[cell] then difference[#difference + 1] = cell end
                            end
                            if applyMineDeduction(simulation, difference, large.remaining - small.remaining) then
                                changed = true
                                break
                            end
                        end
                    end
                end
                if changed then break end
            end
        end
        if not changed then return false end
    end
    return true
end

local function playMineCell(field, cell)
    if field.mines[cell] then
        playChatSound("explosion")
        field.result = "Mine hit. Press R to try again."
        return
    end
    revealMineArea(field, cell)
    if field.opened == MINE_CELLS - field.count then
        field.result = "You cleared the minefield! Press R to play again."
    end
end

local function revealMineCell()
    local field = minefield
    local cell = field.cursor
    if field.result or field.flags[cell] or field.open[cell] or field.generating then return end
    if field.started then
        playMineCell(field, cell)
    elseif not field.noGuess then
        field.mines, field.started = generateMines(cell, field.count), true
        playMineCell(field, cell)
    else
        if mineSearch then mineSearch() end
        field.generating, field.searchFailed = true, false
        mineSearch = coroutine.wrap(function()
            local function checkpoint()
                coroutine.yield(true)
                return minefield == field
            end
            for _ = 1, 120 do
                if minefield ~= field then return end
                local candidate = generateMines(cell, field.count)
                if solvesWithoutGuessing(candidate, cell, field.count, checkpoint) then
                    field.mines, field.started, field.generating = candidate, true, false
                    playMineCell(field, cell)
                    return
                end
                if not checkpoint() then return end
            end
            field.generating, field.searchFailed = false, true
        end)
    end
end

local function handleMines(key, enter)
    local field = minefield
    if not field then
        if key == 17 then mineDifficulty = math.max(1, mineDifficulty - 1)
        elseif key == 18 then mineDifficulty = math.min(#MINE_DIFFICULTIES, mineDifficulty + 1)
        elseif key == 32 then mineNoGuess = not mineNoGuess
        elseif enter then minefield = newMinefield() end
        return
    end
    local cell = field.cursor
    if key == 68 or key == 100 then
        minefield = nil
    elseif key == 82 or key == 114 then
        minefield = newMinefield()
    elseif field.generating then
        return
    elseif key == 17 and cell > MINE_SIDE then
        field.cursor = cell - MINE_SIDE
    elseif key == 18 and cell <= MINE_SIDE * (MINE_SIDE - 1) then
        field.cursor = cell + MINE_SIDE
    elseif key == 19 and (cell - 1) % MINE_SIDE > 0 then
        field.cursor = cell - 1
    elseif key == 20 and cell % MINE_SIDE > 0 then
        field.cursor = cell + 1
    elseif enter then
        revealMineCell()
    elseif (key == 70 or key == 102) and not field.result and not field.open[cell] then
        if field.flags[cell] then
            field.flags[cell] = nil
            field.flagCount = field.flagCount - 1
        elseif field.flagCount < field.count then
            field.flags[cell] = true
            field.flagCount = field.flagCount + 1
        end
    end
end

    -- Collision and ray traversal
    -- DDA visits grid cells in depth order. A sliding door intersects at its center plane.
    local function cast(x, y, dx, dy, view)
        local gx, gy = floor(x), floor(y)
        local sx, sy = dx < 0 and -1 or 1, dy < 0 and -1 or 1
        local ix, iy = dx == 0 and math.huge or abs(1 / dx), dy == 0 and math.huge or abs(1 / dy)
        local tx = (sx < 0 and x - gx or gx + 1 - x) * ix
        local ty = (sy < 0 and y - gy or gy + 1 - y) * iy
        if dx == 0 then tx = math.huge end
        if dy == 0 then ty = math.huge end
        local barrier, barrierU, barrierSide
        while true do
            local d, side
            if tx < ty then
                d, side, gx, tx = tx, 1, gx + sx, tx + ix
            else
                d, side, gy, ty = ty, 2, gy + sy, ty + iy
            end
            local cell = tile(gx, gy)
            local u = side == 1 and y + d * dy or x + d * dx
            if cell == 1 then return d, side, cell, u-floor(u), gx, gy, barrier, barrierU, barrierSide end
            if cell == 3 and view and not barrier then barrier,barrierU,barrierSide=d,u-floor(u),side end
            if cell == 2 then
                local door = doors[index(gx,gy)]
                local normal = door.slideX and dy or dx
                if normal ~= 0 and door.open < 1 then
                    local plane = door.slideX and (gy + 0.5 - y) or (gx + 0.5 - x)
                    local depth = plane / normal
                    local slide = door.slideX and (x + depth * dx - gx) or (y + depth * dy - gy)
                    if depth >= d and depth <= min(tx, ty) and slide >= door.open and slide <= 1 then
                        return depth, door.slideX and 2 or 1, cell, slide-door.open, gx, gy, barrier, barrierU, barrierSide
                    end
                end
            end
        end
    end

    local function clear(x, y)
        local r = 0.19
        for gy = floor(y-r), floor(y+r) do
            for gx = floor(x-r), floor(x+r) do
                local t = tile(gx,gy)
                if t == 1 or t == 3 or (t == 2 and doors[index(gx,gy)].open < 1) then return false end
                local crate=crateCells[index(gx,gy)]
                if crate and abs(x-crate.x)<r+0.28 and abs(y-crate.y)<r+0.28 then return false end
            end
        end
        return true
    end

    local function useDoor(dx, dy)
        for i = 1, 12 do
            local x, y = floor(px + dx * i * 0.125), floor(py + dy * i * 0.125)

            if t == 1 or t == 3 then return end
            if t == 2 then
                local door = doors[index(x,y)]
                if door.target == 0 then
                    door.target = 1
                    effect("doors/door_metal_medium_open1.wav")
                else
                    if abs(px-x-0.5) < 0.72 and abs(py-y-0.5) < 0.72 then return end
                    for _, actor in ipairs(enemies) do
                        if actor.hp > 0 and abs(actor.x-x-0.5) < 0.72 and abs(actor.y-y-0.5) < 0.72 then return end
                    end
                    door.target = 0
                    effect("doors/door_metal_medium_close1.wav")
                end
                return
            end
        end
    end

    local function updateGibs(dt)

            else

                if clear(x,y) then
                    gib.x,gib.y=x,y
                else
                    gib.dx,gib.dy=-gib.dx*0.32,-gib.dy*0.32
                end
                gib.dz=gib.dz-8*dt
                gib.lift=max(0,gib.lift+gib.dz*dt)
                if gib.lift==0 then
                    gib.dz=max(0,-gib.dz*0.27-0.13)

            end
        end

                chooseSupplies(actor.item)
                actor.item.sealed=false
                effect("physics/wood/wood_box_break1.wav",0.55)
                return
            end
            remaining=remaining-1
            if rocket then gibEnemy(actor) end
            local kind=actor.typeIndex or 1
            if kind==10 then explosionSound(0.37)
            else effect(kind<=3 and "npc/metropolice/die1.wav" or "npc/combine_soldier/die1.wav",0.37) end
        end
    end

    local function targetAt(x, y, dx, dy, range, radius)
        local wall,_,_,_,_,_,barrier=cast(x,y,dx,dy,true)
        local nearest, target=min(range,wall),nil
        for kind=1,2 do
            local list=kind==1 and enemies or crates
            for _, actor in ipairs(list) do
                if actor.hp > 0 then
                    local ax, ay = actor.x-x, actor.y-y
                    local depth = ax*dx+ay*dy
                    if depth>0 and depth<nearest and abs(ax*dy-ay*dx)<radius
                        and not (actor.ducking and barrier and barrier<depth) then
                        nearest, target = depth, actor
                    end
                end
            end
        end
        return target
    end

    local function launch(x,y,dx,dy,speed,damage,friendly,blast,kind)
        if #projectiles >= 24 then return false end
        projectiles[#projectiles+1] = {x=x,y=y,dx=dx,dy=dy,speed=speed,damage=damage,
            friendly=friendly,blast=blast,kind=kind or (friendly and "rocket" or blast and "orb" or "bolt"),life=3}
        return true
    end

    local function fire(dx, dy, now)
        local w = weapons[selected]
        if w.ammo and ammo[w.ammo] == 0 then
            shot = now+0.25
            notice("Out of ammo")
            effect("weapons/ar2/ar2_empty.wav",0.38)
            return
        end
        local damage = w.damage*damageBoost
        local cue=w.sound
        if selected == 5 then
            if not launch(px,py,dx,dy,7,damage,true,true) then return end
        elseif selected == 7 then
            if not launch(px,py,dx,dy,12,damage,true,false,"crossbowBolt") then return end
        elseif selected == 4 then
            for i = -2, 2 do
                local angle = yaw+i*0.078
                local target = targetAt(px,py,math.cos(angle),math.sin(angle),9,0.26)
                if target then hit(target,damage) end
            end
        else
            local target = targetAt(px,py,dx,dy,w.range or 18,0.29)
            if target then hit(target,damage)
            elseif selected==1 then cue="weapons/iceaxe/iceaxe_swing1.wav" end
        end
        effect(cue,0.62)
        if w.ammo then ammo[w.ammo] = ammo[w.ammo]-1 end
        shot = now+w.delay/fireRate
        return w.noise
    end

    local function explodeRocket(p,target)
        explosionSound(0.56)
        for kind=1,2 do
            local list=kind==1 and enemies or crates
            for _, actor in ipairs(list) do
                local ax, ay = actor.x-p.x, actor.y-p.y
                local range = ax*ax+ay*ay
                if (actor.hp>0 or (actor.kind=="guard" and not actor.gibbed))
                    and (actor==target or (range<2.25 and (range<0.001 or cast(p.x,p.y,ax,ay)>=1))) then
                    if actor.hp>0 then
                        hit(actor,actor==target and p.damage or p.damage*(1-math.sqrt(range)/1.65),true)
                    else
                        gibEnemy(actor)
                    end
                end
            end
        end
    end

    local function pierceTargets(p,distance,barrier)
        p.hit=p.hit or {}
        for kind=1,2 do
            local list=kind==1 and enemies or crates
            for _,actor in ipairs(list) do
                if actor.hp>0 and not p.hit[actor] then
                    local dx,dy=actor.x-p.x,actor.y-p.y
                    local depth=dx*p.dx+dy*p.dy
                    local covered=p.crossedBarrier or (barrier and barrier<depth)
                    if depth>=0 and depth<=distance and abs(dx*p.dy-dy*p.dx)<0.31
                        and not (actor.ducking and covered) then
                        p.hit[actor]=true
                        hit(actor,p.damage)
                        effect(kind==1 and "weapons/crossbow/hitbod1.wav" or "weapons/crossbow/hit1.wav",0.58)
                    end
                end
            end
        end
        if barrier and barrier<=distance then p.crossedBarrier=true end
    end

    local function updateProjectiles(dt)
        for i = #projectiles, 1, -1 do
            local p = projectiles[i]
            local travel = p.speed*dt
            local piercing=p.kind=="crossbowBolt"
            local wall,_,_,_,_,_,barrier = cast(p.x,p.y,p.dx,p.dy,piercing or not p.friendly)
            if barrier and not p.friendly then wall=min(wall,barrier) end
            local distance = min(travel,max(0,wall-0.01))
            local target
            if piercing then
                pierceTargets(p,distance,barrier)
            elseif p.friendly then
                target=targetAt(p.x,p.y,p.dx,p.dy,travel+0.13,0.31)
            end
            p.x, p.y, p.life = p.x+p.dx*distance, p.y+p.dy*distance, p.life-dt
            local nearPlayer = (px-p.x)^2+(py-p.y)^2
            local collision = wall <= travel or target or (not p.friendly and nearPlayer < 0.11)
            if collision or p.life <= 0 then
                if piercing then
                    if collision then effect("weapons/crossbow/hit1.wav",0.52) end
                elseif p.friendly then
                    explodeRocket(p,target)
                else
                    if p.blast then explosionSound(0.5) end
                    if nearPlayer<0.11 then
                        hurt(p.damage)
                    elseif p.blast and nearPlayer<0.8 then
                        local depth,_,_,_,_,_,cover=cast(p.x,p.y,px-p.x,py-p.y,true)
                        if depth>=1 and (not cover or cover>=1) then hurt(p.damage) end
                    end
                end
                table.remove(projectiles,i)
            end
        end
    end

    local function collect()
        for _, item in ipairs(pickups) do
            if not item.taken and not item.sealed and abs(px-item.x) < 0.43 and abs(py-item.y) < 0.43 then
                local kind = item.kind
                if kind == "key" then hasKey=true; notice("Exit keycard acquired")
                elseif kind == "weapon" then
                    local w = weapons[item.value]
                    owned[item.value], selected = true, item.value
                    giveAmmo(w.ammo,ammoTypes[w.ammo].starter)
                    notice(w.name)
                elseif kind == "health" and health < maxHealth then health=min(maxHealth,health+30); notice("+ Health")
                elseif kind == "armor" and health==maxHealth and armor < 150 then armor=min(150,armor+30); notice("Suit battery +30 armor")
                elseif kind == "ammo" then
                    local k=item.value
                    local box=ammoTypes[k].box
                    if giveAmmo(k,floor(box*supplyBoost)) then notice("+ Ammo"); item.taken=true; effect(pickupSounds.ammo) end
                    kind=nil
                else kind=nil end
                if kind then item.taken=true; effect(pickupSounds[kind]) end
            end
        end
    end

    -- Enemy movement, hearing, and attacks
    local function enemyShot(actor, now, cue, volume, pitch)
        actor.flashUntil=now+0.085
        effect(cue,volume,pitch)
    end

    local function enemyHitscan(damage, covered)
        if not covered or math.random(3)==1 then hurt(damage) end
    end

    local function moveEnemy(actor,x,y,distance)
        local dx,dy=x-actor.x,y-actor.y
        local length=math.sqrt(dx*dx+dy*dy)
        if length<0.001 then return false end
        local step=min(1,distance/length)
        local beforeX,beforeY=actor.x,actor.y
        local nx,ny=beforeX+dx*step,beforeY+dy*step
        if clear(nx,beforeY) then actor.x=nx end
        if clear(actor.x,ny) then actor.y=ny end
        return actor.x~=beforeX or actor.y~=beforeY
    end

    local function buildRoute(goal, limit, hearing)
        local field,queue={[goal]=goal},{goal}
        local i,edge,depth=1,1,0
        while i<=#queue and (not limit or depth<limit) do
            local cell=queue[i]
            local x,y=cell%SIZE,floor(cell/SIZE)
            for _,dir in ipairs(directions) do
                local neighbor=index(x+dir[1],y+dir[2])
                local kind=grid[neighbor]
                if not field[neighbor] and (hearing or not crateCells[neighbor]) then
                    if kind==0 or (kind==2 and doors[neighbor].open==1) or (hearing and kind==3) then
                        field[neighbor]=cell
                        queue[#queue+1]=neighbor
                    end
                end
            end
            if i==edge then edge,depth=#queue,depth+1 end
            i=i+1
        end
        return field
    end

    local function hearGunshot(radius,now)
        local goal=index(floor(px),floor(py))
        if not soundField or soundField.goal~=goal or soundField.radius~=radius or soundField.doors~=doorRevision then
            soundField={goal=goal,radius=radius,doors=doorRevision,audible=buildRoute(goal,radius,true)}
        end
        soundField.x,soundField.y=px,py
        for _,actor in ipairs(enemies) do
            if actor.hp>0 and soundField.audible[index(floor(actor.x),floor(actor.y))] then
                actor.heard,actor.heardUntil=soundField,now+15
            end
        end
    end

    local function stepRoute(actor, field, goal, gx, gy, distance)
        local x,y=floor(actor.x),floor(actor.y)
        local cell=index(x,y)
        local nextCell=field[cell]
        if not nextCell then return end
        if cell==goal then moveEnemy(actor,gx,gy,distance); return end
        local nx,ny=nextCell%SIZE+0.5,floor(nextCell/SIZE)+0.5
        if nx~=x+0.5 and abs(actor.y-(y+0.5))>0.20 then nx=actor.x
        elseif ny~=y+0.5 and abs(actor.x-(x+0.5))>0.20 then ny=actor.y end
        moveEnemy(actor,nx,ny,distance)
    end

    local function followRoute(actor, distance)
        local goal=index(floor(px),floor(py))
        if not routeField or routeOrigin~=goal then
            routeField,routeOrigin=buildRoute(goal),goal
        end
        stepRoute(actor,routeField,goal,px,py,distance)
    end

    local function investigate(actor, distance, now)
        local heard=actor.heard
        if not heard then return false end
        if now>=actor.heardUntil or (actor.x-heard.x)^2+(actor.y-heard.y)^2<0.08 then
            actor.heard=nil
            return false
        end
        if not heard.route or heard.revision~=navigationRevision then
            heard.route=buildRoute(heard.goal,heard.radius+8)
            heard.revision=navigationRevision
        end
        stepRoute(actor,heard.route,heard.goal,heard.x,heard.y,distance)
        return true
    end

    local function findCover(actor, range)
        local ax,ay=actor.x,actor.y
        local startX,startY=floor(ax),floor(ay)
        local best=8
        for y=max(1,startY-2),min(SIZE-2,startY+2) do
            for x=max(1,startX-2),min(SIZE-2,startX+2) do
                if coverCells[index(x,y)] then
                    local cx,cy=x+0.5,y+0.5
                    local dx,dy=cx-ax,cy-ay
                    local distance=dx*dx+dy*dy
                    local gx,gy=px-cx,py-cy
                    if distance>0.04 and distance<best and gx*gx+gy*gy<range*range and clear(cx,cy)
                        and clear(ax+dx*0.5,ay+dy*0.5) then
                        local route,_,_,_,_,_,obstacle=cast(ax,ay,dx,dy,true)
                        if route>1 and (not obstacle or obstacle>1) then
                            local sight,_,_,_,_,_,barrier=cast(cx,cy,gx,gy,true)
                            if sight>1 and barrier and barrier<1 then
                                actor.coverX,actor.coverY=cx,cy
                                best=distance
                            end
                        end
                    end
                end
            end
        end
    end

    local function updateEnemies(dt,now)
        for _, actor in ipairs(enemies) do
            if actor.hp > 0 then
                local stats=actor.stats
                local gx, gy = px-actor.x, py-actor.y
                local squared = gx*gx+gy*gy
                local mode=stats[4]
                local range=stats[5]
                local enraged=mode=="boss" and actor.hp<=actor.maxHp*0.5
                actor.ducking=false
                local sight,cover = 0,nil
                if squared<max(64,range*range) and squared>0.001 then
                    local depth,_,_,_,_,_,barrier=cast(actor.x,actor.y,gx,gy,true)
                    sight,cover=depth,barrier
                end
                if sight>1 then
                    if not actor.lastSight or now-actor.lastSight>2 then
                        actor.alertUntil=now+0.9
                        actor.attack=max(actor.attack,actor.alertUntil)
                        if now>=spottedUntil then spottedUntil=actor.alertUntil end
                        actor.reportUntil=now+2.4
                    end
                    if actor.reportUntil then
                        if now>=actor.reportUntil then
                            actor.reportUntil=nil
                        elseif now>=nextVoiceAt or (actor.typeIndex==10 and not actor.hasSpoken) then
                            local lines=(actor.typeIndex or 1)<=3 and metroAlerts or soldierAlerts
                            effect(actor.typeIndex==10 and breenAlert or lines[math.random(#lines)],0.76)
                            nextVoiceAt,actor.hasSpoken,actor.reportUntil=now+1.05,true,nil
                        end
                    end
                    actor.lastSight=now
                    local d = math.sqrt(squared)
                    local covered=cover and cover<1
                    local inCover=covered and cover*d<=1.35
                    local seek=mode~="melee" and mode~="boss"
                    if mode=="melee" then actor.pursueUntil=now+5 end
                    if inCover then actor.coverX=nil end
                    if seek and not inCover and not actor.windup and not actor.burst
                        and not actor.coverX and now>=(actor.checkCoverAt or 0) then
                        actor.checkCoverAt=now+1.4
                        findCover(actor,range)
                        if actor.coverX then actor.coverUntil=now+3.5 end
                    end
                    if actor.coverX and now>=actor.coverUntil then actor.coverX=nil end
                    if actor.coverX then
                        if not moveEnemy(actor,actor.coverX,actor.coverY,dt*stats[3]) then actor.coverX=nil end
                    elseif d <= range then
                        if actor.burst and now>=actor.attack then
                            enemyShot(actor,now,enemySounds[actor.typeIndex or 2],0.34)
                            enemyHitscan(stats[6],covered)
                            actor.burst=actor.burst-1
                            actor.attack=now+(actor.burst>0 and 0.12 or stats[7])
                            if actor.burst==0 then actor.burst=nil end
                        elseif not actor.burst and not actor.windup and now >= actor.attack then
                            actor.windup=now+(mode=="melee" and 0.27 or 0.42)
                        elseif actor.windup and now >= actor.windup then
                            actor.windup, actor.attack = nil, now+stats[7]
                            local enemySound=enemySounds[actor.typeIndex or 2]
                            if mode=="burst" then
                                enemyShot(actor,now,enemySound,0.36)
                                enemyHitscan(stats[6],covered)
                                actor.burst, actor.attack=2, now+0.12
                            elseif mode=="elite" then
                                if d>=4 and math.random(3)==1 then
                                    if launch(actor.x,actor.y,gx/d,gy/d,4.8,34,false,true) then
                                        enemyShot(actor,now,"weapons/ar2/npc_ar2_altfire.wav",0.48)
                                    end
                                else
                                    enemyShot(actor,now,enemySound,0.4)
                                    enemyHitscan(stats[6],covered)
                                end
                            elseif mode=="boss" then
                                local heading=math.atan2(gy,gx)
                                local count=enraged and 2 or 1
                                local fired=false
                                for i=-count,count do
                                    local angle=heading+i*0.14
                                    if launch(actor.x,actor.y,math.cos(angle),math.sin(angle),4.2,stats[6],false,true) then fired=true end
                                end
                                if fired then enemyShot(actor,now,enemySound,0.62,80) end
                                actor.attack=now+(enraged and 0.77 or stats[7])
                            else
                                if mode=="melee" then
                                    effect(enemySound,0.45)
                                    if not covered then hurt(stats[6]) end
                                else
                                    enemyShot(actor,now,enemySound,0.45)
                                    enemyHitscan(stats[6],covered)
                                end
                            end
                        end
                    else
                        actor.windup, actor.burst=nil,nil
                        local dist=dt*stats[3]
                        if mode=="melee" and (covered or not moveEnemy(actor,px,py,dist)) then
                            followRoute(actor,dist)
                        elseif mode~="melee" then
                            moveEnemy(actor,px,py,dist)
                        end
                    end
                    actor.ducking=seek and inCover and d<=range and not actor.windup and not actor.burst
                        and now>=(actor.flashUntil or 0)
                else
                    actor.windup,actor.burst,actor.coverX=nil,nil,nil
                    local checkingSound=investigate(actor,dt*stats[3],now)
                    if not checkingSound and mode=="melee" and squared>0.001 and now<(actor.pursueUntil or 0) then
                        followRoute(actor,dt*stats[3])
                    end
                end
            end
        end
    end

    -- Input and simulation
    local function pollPress(action, key)
        local held = down(action, key)
        if held and not previous[action] then pendingPress[action]=true end
        previous[action] = held
    end

    local function takePress(action)
        local edge=pendingPress[action]
        pendingPress[action]=nil
        return edge
    end

    local muzzleUntil = 0
    function game.update()
        pollPress("Automap",KEY.TAB)
        pollPress("Restart",KEY.R)
        pollPress("Use",KEY.E)
        pollPress("Menu",KEY.M)
        for i=1,#slotActions do pollPress(slotActions[i],slotKeys[i]) end
        local realtime=timer.curtime()
        if lastTime and realtime-lastTime < 1/60 then return end
        local dt=min(realtime-(lastTime or realtime),0.05)
        lastTime=realtime
        soundSlots=6
        local use=takePress("Use")
        local toggleMenu=takePress("Menu")
        if menu then
            takePress("Automap")
            takePress("Restart")
            for _,action in ipairs(slotActions) do takePress(action) end
            if menu=="title" then
                menuSeed=(menuSeed*48271+floor(timer.systime()*1000000)%2147483647)%2147483646+1
                if use then
                    startRun(menuSeed)
                    menu,musicWanted=nil,true
                    updateMusic(gameTime)
                end
            elseif use or toggleMenu then
                menu=nil
            end
            return
        end
        if toggleMenu then
            menu="pause"
            if music then trackIndex=(trackIndex+#combatTracks-2)%#combatTracks+1 end
            stopMusic()
            return
        end
        gameTime=gameTime+dt
        local now=gameTime
        if takePress("Automap") then map=not map end
        if takePress("Restart") and health==0 then
            startRun(seed%2147483646+1)
            map,muzzleUntil=false,0
        end
        local perkChoice
        for i=1,#weapons do
            if takePress(slotActions[i]) then
                if won and i<=3 then perkChoice=perkChoice or i
                elseif not won and health>0 and owned[i] then selected=i end
            end
        end
        if perkChoice then
            perks[choices[perkChoice]][3]()
            floorNumber=floorNumber+1
            generate(seed%2147483646+1)
            map, muzzleUntil=false, 0
            return
        end
        if health<=0 or won then return end
        local turn=(down("TurnRight",KEY.RIGHT) and 1 or 0)-(down("TurnLeft",KEY.LEFT) and 1 or 0)
        local move=(down("Forward",KEY.UP,KEY.W) and 1 or 0)-(down("Backward",KEY.DOWN,KEY.S) and 1 or 0)
        local strafe=(down("StrafeRight",KEY.D) and 1 or 0)-(down("StrafeLeft",KEY.A) and 1 or 0)
        yaw=(yaw+turn*dt*2.5)%(2*math.pi)
        local attack=down("Fire",KEY.SPACE)
        if turn~=0 or move~=0 or strafe~=0 or use or attack then musicWanted=true end
        updateMusic(now)
        local dx,dy=math.cos(yaw),math.sin(yaw)
        local step=dt*2.8*min(speedBoost,2)
        if move~=0 and strafe~=0 then step=step/math.sqrt(2) end
        local nx,ny=px+(dx*move-dy*strafe)*step,py+(dy*move+dx*strafe)*step
        if clear(nx,py) then px=nx end
        if clear(px,ny) then py=ny end
        if use then
            if abs(px-exitX)<1 and abs(py-exitY)<1 then
                if boss and boss.hp>0 then
                    notice("Defeat Mecha Breen to unlock the exit")
                elseif hasKey then
                    won=true
                    stopMusic()
                    effect("buttons/button14.wav")
                    local first=(seed+floorNumber*4)%#perks
                    choices={first+1,(first+2)%#perks+1,(first+4)%#perks+1}
                    return
                else
                    notice("Find the exit keycard")
                end
            else useDoor(dx,dy) end
        end
        for _,door in pairs(doors) do
            if door.open~=door.target then
                local wasOpen=door.open==1
                if door.target==1 then door.open=min(1,door.open+dt*2.7)
                else door.open=max(0,door.open-dt*2.7) end
                if wasOpen~=(door.open==1) then
                    routeField=nil
                    navigationRevision,doorRevision=navigationRevision+1,doorRevision+1
                end
            end
        end
        collect()
        if attack and now>=shot then
            if not weapons[selected].ammo or ammo[weapons[selected].ammo]>0 then muzzleUntil=now+0.075 end
            local noise=fire(dx,dy,now)
            if noise then hearGunshot(noise,now) end
        end
        updateEnemies(dt,now)
        updateProjectiles(dt)
        updateGibs(dt)
        if health==0 then stopMusic() end
    end

                local center = WIDTH/2 + projection * (y*dx-x*dy) / z

local keyboard = {focused = false, repeats = {}, confirmedUntil = 0, nextPing = 0}
setupPermissionRequest({"input", "input.lockcontrols"}, "Control the demo from its linked screen", true)

function keyboard.release()
    if keyboard.focused then
        pcall(input.lockControls, false)
        if net.getBytesLeft() >= 64 then
            net.start(CONTROL_MESSAGE)
            net.writeUInt(RELEASE, 2)
            net.send()
        end
    end
    keyboard.focused, keyboard.repeats = false, {}
end

function keyboard.focus()
    if keyboard.focused or not hasPermission("input") or not hasPermission("input.lockcontrols") then return end
    local ok, allowed = pcall(input.canLockControls)
    if ok and allowed then keyboard.focused = pcall(input.lockControls, true) end
end

function keyboard.sendKey(key)
    if net.getBytesLeft() < 64 then return end
    net.start(CONTROL_MESSAGE)
    net.writeUInt(KEY_PRESS, 2)
    net.writeUInt(key, 8)
    net.send()
end

local function handleKey(key)
    local enter = key == 10 or key == 13 or key == 142
    if not themeConfigured then
        if key == 17 then selectedTheme = math.max(1, selectedTheme - 1)
        elseif key == 18 then selectedTheme = math.min(#themes, selectedTheme + 1)
        elseif enter then applyTheme() end
    elseif key == 163 then
        toggleDesktop()
    elseif gameTab == "free_man" then
        offline.freeMan.key(key, enter)
    elseif gameTab == "music" then
        if enter then offline.player.play()
        elseif key == 32 then offline.player.stop()
        elseif key == 17 then offline.player.move(-1)
        elseif key == 18 then offline.player.move(1)
        elseif key == 151 then offline.player.move(-4)
        elseif key == 152 then offline.player.move(4) end
    elseif gameTab == "paint" then
        offline.paint.key(key, enter)
    elseif gameTab == "squadron" then
        offline.squadron.key(key, enter)
    elseif gameTab == "calculator" then
        calculatorKey(key, enter)
    elseif gameTab == "mines" then
        if key == 8 or key == 127 then gameTab = "menu"
        else handleMines(key, enter) end
    elseif selectedGame == 0 then
        if key == 19 then selectMenuTab(APPS)
        elseif key == 20 then selectMenuTab(GAMES)
        elseif key == 18 or enter then selectedGame = 1 end
    elseif key == 19 and selectedGame % 2 == 0 then
        selectedGame = selectedGame - 1
    elseif key == 20 and selectedGame % 2 == 1 and menuOptions[selectedGame + 1] then
        selectedGame = selectedGame + 1
    elseif key == 17 then
        selectedGame = math.max(0, selectedGame - 2)
    elseif key == 18 and menuOptions[selectedGame + 2] then
        selectedGame = selectedGame + 2
    elseif enter then
        openApp()
    end
end

            keyboard.release()

    if not keyboard.focused then return end
    if button == KEY.LALT or button == KEY.RALT or button == KEY.ESCAPE then
        keyboard.release()
        return
    end
    local key = keyboard.keys[button]
    if key and poweredOn and timer.curtime() >= startupUntil then
        keyboard.sendKey(key)
        if key ~= 163 then keyboard.repeats[button] = timer.curtime() + (keyboard.movement[key] and 0.06 or 0.3) end
    end
end)

    keyboard.repeats[button] = nil
end)

local freeManActive = false
hook.add("think", "demo_games", function()
    local now = timer.curtime()
    if keyboard.focused and now >= keyboard.confirmedUntil then keyboard.release() end
    if keyboard.focused and now >= keyboard.nextPing and net.getBytesLeft() >= 64 then
        keyboard.nextPing = now + 1
        net.start(CONTROL_MESSAGE)
        net.writeUInt(KEEPALIVE, 2)
        net.send()
    end
    if keyboard.focused and poweredOn and now >= startupUntil then
        for button, nextAt in pairs(keyboard.repeats) do
            local key = keyboard.keys[button]
            if not input.isKeyDown(button) then
                keyboard.repeats[button] = nil
            elseif now >= nextAt then
                keyboard.repeats[button] = now + 0.06
                keyboard.sendKey(key)
            end
        end
    end
    if player() ~= owner() then return end
    local active = poweredOn and gameTab == "free_man" and now >= startupUntil
    if active then
        offline.freeMan.update()
    elseif freeManActive then
        offline.freeMan.pause()
    end
    freeManActive = active
    if poweredOn and gameTab == "squadron" and timer.curtime() >= startupUntil then
        offline.squadron.update(timer.curtime())
    else
        offline.squadron.lastTick = nil
    end
    if poweredOn and gameTab == "mines" and mineSearch and not mineSearch() then mineSearch = nil end
end)

local function clickScreen(x, y)
    if x < 16 or x >= 496 then return end
    if not themeConfigured then
        local offset = y - THEME_TOP
        if offset >= 0 and offset < #themes * THEME_STEP and offset % THEME_STEP < THEME_HEIGHT then
            selectedTheme = math.floor(offset / THEME_STEP) + 1
        elseif y >= 419 and y < 466 then applyTheme() end
    elseif x >= 369 and y >= 8 and y < 39 then
        toggleDesktop()
    elseif gameTab == "music" then
        offline.player.click(x, y)
    elseif gameTab == "paint" then
        offline.paint.click(x, y)
    elseif gameTab == "squadron" then
        offline.squadron.key(13, true)
    elseif gameTab == "free_man" then
        offline.freeMan.key(13, true)
    elseif gameTab == "calculator" then
        calculatorClick(x, y)
    elseif gameTab == "mines" then
        if not minefield and y >= 123 and y < 283 and (y - 123) % 57 < 46 then
            mineDifficulty = math.floor((y - 123) / 57) + 1
            minefield = newMinefield()
        elseif not minefield and y >= 302 and y < 338 then
            mineNoGuess = not mineNoGuess
        elseif minefield and x >= 96 and x < 416 and y >= 111 and y < 431 then
            minefield.cursor = math.floor((y - 111) / 40) * MINE_SIDE + math.floor((x - 96) / 40) + 1
            revealMineCell()
        end
    elseif y >= 82 and y < 114 then
        if x < 248 then selectMenuTab(APPS)
        elseif x >= 264 then selectMenuTab(GAMES) end
    elseif (x < 248 or x >= 264) and y >= LAUNCHER_TOP then
        local row, gap = math.floor((y - LAUNCHER_TOP) / 100), (y - LAUNCHER_TOP) % 100
        local index = row * 2 + (x < 248 and 1 or 2)
        if gap < TILE_HEIGHT and menuOptions[index] then
            selectedGame = index
            openApp()
        end
    end
end

net.receive(CONTROL_MESSAGE, function()
    local action = net.readUInt(2)
    if action == KEEPALIVE then
        keyboard.confirmedUntil = timer.curtime() + 3
        keyboard.focus()
    elseif player() == owner() and poweredOn and timer.curtime() >= startupUntil then
        if action == CLICK then
            local x = net.readUInt(9)
            clickScreen(x, net.readUInt(9))
        elseif action == KEY_PRESS then
            handleKey(net.readUInt(8))
        end
    end
end)

local function sendScreenClick(screen, x, y)
    if not x or not y or x < 0 or x >= 512 or y < 0 or y >= 512 or net.getBytesLeft() < 64 then return end
    net.start(CONTROL_MESSAGE)
    net.writeUInt(CLICK, 2)
    net.writeUInt(screen:entIndex(), 16)
    net.writeUInt(math.floor(x), 9)
    net.writeUInt(math.floor(y), 9)
    net.send()
end

hook.add("StarfallUsed", "demo_controls", function(user, screen)
    if not poweredOn or timer.curtime() < startupUntil or user ~= player() or not screen
        or not screen:isValid() or screen:getClass() ~= "starfall_screen" then return end
    sendScreenClick(screen, render.cursorPos(user, screen))
end)

local mouseButtons = {MOUSE.LEFT, MOUSE.RIGHT}
local mouseDown = {}
local pointerX, pointerY
hook.add("think", "demo_screen_mouse", function()
    pointerX, pointerY = nil, nil
    if not hasPermission("input") then return end
    local clicked = false
    for _, button in ipairs(mouseButtons) do
        local down = input.isMouseDown(button)
        clicked = clicked or (down and mouseDown[button] == false)
        mouseDown[button] = down
    end
    if not keyboard.focused or not poweredOn or timer.curtime() < startupUntil then return end
    local user = player()
    local screen = user:getEyeTrace().Entity
    if not screen or not screen:isValid() or screen:getClass() ~= "starfall_screen" then return end
    local x, y = render.cursorPos(user, screen)
    if not x or not y or x < 0 or x >= 512 or y < 0 or y >= 512 then return end
    pointerX, pointerY = x, y
    if clicked then sendScreenClick(screen, x, y) end
end)

local function drawPointer()
    if not pointerX then return end
    local x, y = pointerX, pointerY
    display.setColor(Color(0, 0, 0))
    display.drawRect(x - 2, y - 9, 5, 19)
    display.drawRect(x - 9, y - 2, 19, 5)
    display.setColor(Color(255, 255, 255))
    display.drawRect(x, y - 8, 1, 17)
    display.drawRect(x - 8, y, 17, 1)
end

local function drawScreen()
    render.setFont(font)
    if not poweredOn then
        render.setColor(Color(9, 13, 19))
        render.drawRect(0, 0, 512, 512)
        label(22, 449, "Computer off", muted)
        label(22, 479, "Set the Wiremod Power input to 1.", muted, smallFont)
        return
    end
    local now = timer.curtime()
    render.setColor(background)
    render.drawRect(0, 0, 512, 512)
    local crashRemaining = startupUntil - now - STARTUP_DURATION
    if crashRemaining > 0 then
        render.setColor(Color(7, 15, 25))
        render.drawRect(0, 0, 512, 512)
        label(28, 114, "SYSTEM CRASH", Color(255, 120, 125), titleFont)
        local detail = "Calculator exception: division by zero."
        local visible = math.floor(#detail * math.min(1, (2.5 - crashRemaining) / 1.2))
        label(28, 178, detail:sub(1, visible), Color(230, 241, 250), smallFont)
        label(28, 224, "Restarting in " .. math.ceil(crashRemaining) .. "s", Color(155, 176, 198), smallFont)
        return
    end
    if now < startupUntil then
        local progress = 1 - (startupUntil - now) / STARTUP_DURATION
        label(72, 200, "Starting computer", ink, titleFont)
        render.setColor(panel)
        render.drawRect(72, 260, 368, 10)
        if progress > 0 then
            render.setColor(accent)
            render.drawRect(72, 260, 368 * progress, 10)
        end
        return
    end
    render.setColor(panel)
    render.drawRect(0, 0, 512, 72)
    if not themeConfigured then renderThemeSetup(); return end
    if gameTab == "free_man" then
        offline.freeMan.draw()
        render.setColor(panel)
        render.drawRect(369, 8, 127, 31)
        label(380, 15, "F1 Desktop", accent, smallFont)
        if keyboard.focused then label(380, 43, "Alt releases keys", ink, smallFont) end
        return
    end
    local title = gameTab == "calculator" and "Calculator" or (gameTab == "mines" and "Minesweeper" or "Computer Demo")
    if gameTab == "music" then title = "HL2 Music Player"
    elseif gameTab == "paint" then title = "Paint"
    elseif gameTab == "squadron" then title = "Star Squadron" end
    label(22, 10, title, ink, titleFont)
    label(22, 46, "Offline demo", muted, smallFont)
    if gameTab ~= "menu" or lastApp then
        render.setColor(background)
        render.drawRect(369, 8, 127, 31)
        label(380, 15, gameTab == "menu" and "F1 Last app" or "F1 Desktop", accent, smallFont)
    end
    if gameTab == "calculator" then renderCalculator()
    elseif gameTab == "music" then offline.player.draw()
    elseif gameTab == "paint" then offline.paint.draw()
    elseif gameTab == "squadron" then offline.squadron.draw()
    elseif gameTab == "mines" then renderMines()
    else renderDesktop() end
end

local sharedFrame, nextCapture = nil, 0
net.receive(SCREEN_MESSAGE, function(bits)
    if player() == owner() then return end
    local data = readScreenFrame(bits)
    if not data then return end
    local decoded = bit.decompress(data, 384000)
    if not decoded then return end
    local ok, commands = pcall(json.decode, decoded)
    if ok and type(commands) == "table" then sharedFrame = commands end
end)

hook.add("render", "demo_screen", function()
    if player() ~= owner() then
        if sharedFrame then
            for _, command in ipairs(sharedFrame) do
                local op = command[1]
                if op == 5 then
                    display.setFont(fonts[command[2]])
                elseif op == 6 then
                    display.setColor(Color(command[2], command[3], command[4], command[5]))
                else
                    display[drawMethods[op]](unpack(command, 2))
                end
            end
        else
            display.setColor(Color(9, 13, 19))
            display.drawRect(0, 0, 512, 512)
            label(22, 240, "Waiting for the owner's demo screen...", ink, smallFont)
        end
        drawPointer()
        return
    end

    local now = timer.curtime()
    if bit and bit.compress and json and now >= nextCapture then
        nextCapture = now + 0.2
        frame = {}
    end
    drawScreen()
    if frame then
        local commands = frame
        frame = nil
        local data = bit.compress(json.encode(commands))
        if data and #data <= MAX_FRAME_BYTES then sendScreenFrame(data) end
    end
    drawPointer()
end)
