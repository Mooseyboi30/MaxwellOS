--@name MaxwellNET Website Template
--@shared
--@include maxwellnet_storage.txt

local createStorage = require("maxwellnet_storage.txt")
-- Change this key once for a new site. Keep it stable through future uploads.
local storage = createStorage("website_template")
local NAME = "MaxwellNET Example"
local FONT = {"Roboto", 22, 500}
local events = setmetatable({}, {__mode = "k"})
local nextEvent = 0
local pointer = setmetatable({}, {__mode = "k"})

local function optionsFor(route)
    if route == "notes" then
        return {512, links = {
            {28, 390, 210, 48, "Home", "", "home"},
            {266, 390, 218, 48, "About", "", "about"}
        }}
    elseif route == "about" then
        return {640, links = {{28, 560, 456, 48, "Home", "", "home"}}}
    end
    return {512,
        requires = {"textFields", "actionResults", "subsites"},
        inputs = {{28, 155, 456, 48, "Write a note", "note", 80, ""}},
        buttons = {
            {28, 225, 216, 48, "Save note", "save_note"},
            {268, 225, 216, 48, "Add one", "increment"}
        },
        links = {
            {28, 300, 216, 44, "Saved note", "", "notes"},
            {268, 300, 216, 44, "About", "", "about"}
        }
    }
end

local function frameFor(user, route)
    local frame = {
        {"rect", 0, 0, 512, 512, 21, 32, 48},
        {"text", 28, 28, 456, 34, 245, 250, 255, NAME},
        {"text", 28, 78, 456, 28, 179, 214, 228, storage.status}
    }
    local function text(y, value)
        frame[#frame + 1] = {"text", 28, y, 456, 30, 245, 250, 255, value:sub(1, 96)}
    end
    if not storage.ready then text(120, "Waiting for the owner's saved site data"); return frame end
    if route == "notes" then
        text(145, "Saved note")
        text(195, storage.get("note", "No note saved yet"))
    elseif route == "about" then
        text(140, "Text fields, links, actions, and local storage")
        text(190, "Use Back to return to the previous subsite.")
        text(240, "Hold a mouse button to send pointer events.")
        text(290, "Save adds this route to your Favorites.")
        frame[#frame + 1] = {"rect", 0, 512, 512, 128, 21, 32, 48}
        text(535, "This subsite scrolls. Home is below.")
    else
        text(112, "Count: " .. storage.get("count", 0))
        frame[#frame + 1] = {"rect", 28, 225, 216, 48, 45, 115, 108}
        frame[#frame + 1] = {"text", 42, 237, 188, 30, 245, 250, 255, "Save note"}
        frame[#frame + 1] = {"rect", 268, 225, 216, 48, 60, 76, 112}
        frame[#frame + 1] = {"text", 282, 237, 188, 30, 245, 250, 255, "Add one"}
        text(370, "Scratch / drag test")
        local point = pointer[user]
        local progress = point and point.progress or 0
        frame[#frame + 1] = {"rect", 28, 410, 456, 55, 93, 111, 126}
        frame[#frame + 1] = {"rect", 28, 410, progress * 456, 55, 45, 115, 108}
    end
    return frame
end

if SERVER then
    hook.add("remote", "maxwellnet_example", function(sender, user, protocol, first, second, third, fourth, fifth, sixth)
        if protocol == "starfall_website_discover_v1" then return chip():entIndex(), NAME end
        if protocol == "starfall_website_page_v1" then
            local capabilities, route = first, second or "home"
            -- Older viewers receive a readable fallback page.
            if type(capabilities) ~= "table" or not capabilities.subsites then
                return chip():entIndex(), NAME, NAME, {"Update MaxwellOS to use this example."},
                    false, FONT, false, false, {512}
            end
            return chip():entIndex(), NAME, NAME, {"Text, storage, links, and subsites."},
                false, FONT, false, false, optionsFor(route)
        end
        if not user or not user:isValid() or not user:isPlayer() then return end
        if protocol == "starfall_website_screen_v1" then
            local route = second or "home"
            local frame = frameFor(user, route)
            local event = events[user]
            if event and timer.curtime() >= event.at and timer.curtime() < event.at + 1 then
                return frame, "win", event.token
            end
            return frame
        elseif protocol == "starfall_website_pointer_v1" then
            local x, y, held, right = first, second, third, fourth
            if (sixth or "home") ~= "home" or type(x) ~= "number" or type(y) ~= "number"
                or x < 28 or x > 484 or y < 410 or y > 465 or not (held or right) then return end
            local last = pointer[user]
            local progress = last and last.progress or 0
            if last then progress = math.min(1, progress + math.abs(x-last.x)/456) end
            pointer[user] = {x=x, progress=progress}
        elseif protocol == "starfall_website_input_v1" or protocol == "starfall_website_button_v1" then
            if not storage.ready then return {ok=false, message=storage.status, sound="error"} end
            if protocol == "starfall_website_input_v1" and first == "note" then
                local ok, err = storage.set("note", tostring(second):sub(1,80))
                return {ok=ok, message=ok and "Note queued for saving." or err}
            elseif first == "save_note" then
                local values = type(second) == "table" and second or {}
                local ok, err = storage.set("note", tostring(values.note or ""):sub(1,80))
                return {ok=ok, message=ok and "Note queued for saving." or err, sound=ok and "success" or "error"}
            elseif first == "increment" then
                local ok, err = storage.set("count", storage.get("count",0)+1)
                if not ok then return {ok=false, message=err, sound="error"} end
                nextEvent = nextEvent + 1
                events[user] = {at=timer.curtime()+1, token="count:"..nextEvent}
                return {ok=true, message="Count increased.", sound="click"}
            end
            return {ok=false, message="Unknown action."}
        end
    end)
end
